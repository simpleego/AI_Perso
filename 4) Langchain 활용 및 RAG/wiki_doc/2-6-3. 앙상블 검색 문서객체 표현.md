```python
def create_document(content, file_name, page, category):
    return Document(
        page_content=content,
        metadata={
            "source": file_name,         # 파일명 명시
            "page": page,                # 페이지 명시
            "category": category,        # 체계화된 분류
            "created_at": "2026-07-06"   # 생성일 등 관리 정보
        }
    )

# 사용 시
doc = create_document("벡터 데이터베이스는...", "db_manual.pdf", 12, "db/vector")

```
