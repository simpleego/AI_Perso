```py
"""
Google Colab 실습: LangChain + HuggingFaceEmbeddings

참고: https://wikidocs.net/231573

Colab 실행 순서
1. 아래 [셀 1]의 pip 명령을 별도 코드 셀에서 먼저 실행합니다.
2. 필요하면 런타임 > 런타임 유형 변경 > T4 GPU를 선택합니다.
3. [셀 2] 이하의 파이썬 코드를 순서대로 실행합니다.

이 실습은 로컬 추론 방식이므로 OpenAI/Hugging Face API 키가 필요하지 않습니다.
모델 파일은 첫 실행 때 Hugging Face에서 내려받습니다.
"""

# =============================================================================
# [셀 1] 패키지 설치
# Colab 코드 셀에 다음 한 줄만 복사하여 실행하세요.
# =============================================================================
# !pip -q install -U langchain-huggingface sentence-transformers


# =============================================================================
# [셀 2] 라이브러리 불러오기 및 실행 장치 확인
# =============================================================================
import numpy as np
import torch
from langchain_huggingface import HuggingFaceEmbeddings


device = "cuda" if torch.cuda.is_available() else "cpu"
print(f"실행 장치: {device}")


# =============================================================================
# [셀 3] 한국어 임베딩 모델 생성
# =============================================================================
MODEL_NAME = "jhgan/ko-sroberta-nli"

embeddings_model = HuggingFaceEmbeddings(
    model_name=MODEL_NAME,
    model_kwargs={"device": device},
    encode_kwargs={
        "normalize_embeddings": True,
        "batch_size": 32,
    },
    show_progress=True,
)

print(f"모델 로딩 완료: {MODEL_NAME}")


# =============================================================================
# [셀 4] 여러 문서를 임베딩 벡터로 변환
# =============================================================================
documents = [
    "안녕하세요!",
    "어! 오랜만이에요.",
    "이름이 어떻게 되세요?",
    "날씨가 추워요.",
    "Hello LLM!",
]

document_vectors = embeddings_model.embed_documents(documents)

print(f"문서 개수: {len(document_vectors)}")
print(f"임베딩 차원: {len(document_vectors[0])}")
print("첫 번째 문서 벡터 앞부분:", document_vectors[0][:5])


# =============================================================================
# [셀 5] 질의를 임베딩하고 각 문서와 코사인 유사도 계산
# =============================================================================
query = "첫인사를 하고 이름을 물어봤나요?"
query_vector = embeddings_model.embed_query(query)


def cosine_similarity(vector_a: list[float], vector_b: list[float]) -> float:
    """두 벡터의 코사인 유사도를 반환합니다."""
    a = np.asarray(vector_a, dtype=np.float32)
    b = np.asarray(vector_b, dtype=np.float32)
    denominator = np.linalg.norm(a) * np.linalg.norm(b)

    if denominator == 0:
        return 0.0

    return float(np.dot(a, b) / denominator)


results = [
    {
        "document": document,
        "score": cosine_similarity(document_vector, query_vector),
    }
    for document, document_vector in zip(documents, document_vectors)
]

results.sort(key=lambda item: item["score"], reverse=True)

print(f"\n질의: {query}")
print("-" * 70)

for rank, result in enumerate(results, start=1):
    print(
        f"{rank}위 | 유사도: {result['score']:.4f} | "
        f"문서: {result['document']}"
    )


# =============================================================================
# [셀 6] 가장 유사한 문서 확인
# =============================================================================
best_match = results[0]

print("\n가장 관련성이 높은 문서")
print(f"문서: {best_match['document']}")
print(f"유사도: {best_match['score']:.4f}")


# =============================================================================
# [선택 셀 7] 정규화된 임베딩에서는 내적과 코사인 유사도가 거의 같음
# =============================================================================
document_matrix = np.asarray(document_vectors, dtype=np.float32)
query_array = np.asarray(query_vector, dtype=np.float32)
dot_product_scores = document_matrix @ query_array

print("\n정규화 벡터의 내적 점수")
for document, score in zip(documents, dot_product_scores):
    print(f"{float(score):.4f} | {document}")


```
