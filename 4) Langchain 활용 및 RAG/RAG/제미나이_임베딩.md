```py
"""
Google Colab 실습: LangChain + GoogleGenerativeAIEmbeddings

참고: https://wikidocs.net/231572

실행 순서
1. Google AI Studio(https://aistudio.google.com/app/apikey)에서 API 키를 발급합니다.
2. Colab 왼쪽의 열쇠 모양 '보안 비밀'에서 다음 값을 등록합니다.
   - 이름: GOOGLE_API_KEY
   - 값: 발급받은 API 키
   - 노트북 액세스 허용: 활성화
3. 아래 [셀 1]부터 순서대로 실행합니다.

주의
- API 키를 코드에 직접 입력하거나 공개 저장소에 올리지 마세요.
- 이 파일은 # %% 구분 및 [셀] 주석을 기준으로 Colab에 복사할 수 있습니다.
"""

# =============================================================================
# [셀 1] 최신 패키지 설치
# Colab 코드 셀에서 다음 한 줄을 실행하세요.
# =============================================================================
# !pip -q install -U langchain-google-genai


# =============================================================================
# [셀 2] Google API 키 안전하게 불러오기
# =============================================================================
import os
from getpass import getpass


def load_google_api_key() -> str:
    """Colab 보안 비밀 또는 화면 입력으로 Google API 키를 불러옵니다."""
    api_key = os.getenv("GOOGLE_API_KEY") or os.getenv("GEMINI_API_KEY")

    # Google Colab에서 실행 중이면 보안 비밀을 먼저 확인합니다.
    if not api_key:
        try:
            from google.colab import userdata

            api_key = userdata.get("GOOGLE_API_KEY")
        except (ImportError, KeyError, TypeError):
            api_key = None

    # 보안 비밀이 없을 때만 화면에서 입력받습니다.
    if not api_key:
        api_key = getpass("Google AI API 키를 입력하세요: ").strip()

    if not api_key:
        raise RuntimeError(
            "GOOGLE_API_KEY가 없습니다. "
            "Colab 보안 비밀에 GOOGLE_API_KEY를 등록하세요."
        )

    os.environ["GOOGLE_API_KEY"] = api_key
    return api_key


load_google_api_key()
print("Google API 키 설정 완료")


# =============================================================================
# [셀 3] Gemini 임베딩 모델 생성
# =============================================================================
from langchain_google_genai import GoogleGenerativeAIEmbeddings


MODEL_NAME = "models/gemini-embedding-001"

embeddings_model = GoogleGenerativeAIEmbeddings(
    model=MODEL_NAME,
    task_type="SEMANTIC_SIMILARITY",
    output_dimensionality=768,
)

print(f"임베딩 모델 준비 완료: {MODEL_NAME}")


# =============================================================================
# [셀 4] 여러 문서를 임베딩 벡터로 변환
# =============================================================================
documents = [
    "안녕하세요!",
    "어! 오랜만이에요.",
    "이름이 어떻게 되세요?",
    "날씨가 추워요.",
    "Hello LLM!",
]

document_vectors = embeddings_model.embed_documents(documents)

print(f"문서 개수: {len(document_vectors)}")
print(f"임베딩 차원: {len(document_vectors[0])}")
print("첫 번째 문서 벡터 앞부분:", document_vectors[0][:5])


# =============================================================================
# [셀 5] 질의 임베딩 및 코사인 유사도 계산
# =============================================================================
import numpy as np


def cosine_similarity(vector_a: list[float], vector_b: list[float]) -> float:
    """두 벡터 사이의 코사인 유사도를 계산합니다."""
    a = np.asarray(vector_a, dtype=np.float32)
    b = np.asarray(vector_b, dtype=np.float32)
    denominator = np.linalg.norm(a) * np.linalg.norm(b)

    if denominator == 0:
        return 0.0

    return float(np.dot(a, b) / denominator)


query = "첫인사를 하고 이름을 물어봤나요?"
query_vector = embeddings_model.embed_query(query)

results = [
    {
        "document": document,
        "score": cosine_similarity(document_vector, query_vector),
    }
    for document, document_vector in zip(documents, document_vectors)
]

results.sort(key=lambda item: item["score"], reverse=True)

print(f"\n질의: {query}")
print("-" * 70)

for rank, result in enumerate(results, start=1):
    print(
        f"{rank}위 | 유사도: {result['score']:.4f} | "
        f"문서: {result['document']}"
    )


# =============================================================================
# [셀 6] 가장 관련성이 높은 문서 출력
# =============================================================================
best_match = results[0]

print("\n가장 관련성이 높은 문서")
print(f"문서: {best_match['document']}")
print(f"유사도: {best_match['score']:.4f}")


# =============================================================================
# [선택 셀 7] 다른 질문으로 재사용하는 함수
# =============================================================================
def search_similar_documents(
    user_query: str,
    top_k: int = 3,
) -> list[dict]:
    """사용자 질문과 의미가 가까운 문서를 top_k개 반환합니다."""
    if not 1 <= top_k <= len(documents):
        raise ValueError(f"top_k는 1~{len(documents)} 범위여야 합니다.")

    user_query_vector = embeddings_model.embed_query(user_query)

    ranked_results = [
        {
            "document": document,
            "score": cosine_similarity(vector, user_query_vector),
        }
        for document, vector in zip(documents, document_vectors)
    ]

    ranked_results.sort(key=lambda item: item["score"], reverse=True)
    return ranked_results[:top_k]


new_query = "오늘 기온이 낮나요?"
print(f"\n새로운 질의: {new_query}")

for rank, result in enumerate(search_similar_documents(new_query), start=1):
    print(
        f"{rank}위 | 유사도: {result['score']:.4f} | "
        f"문서: {result['document']}"
    )

```
