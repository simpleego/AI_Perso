```python

import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt

# 1. 오디오 로드 (샘플 데이터 사용)
y, sr = librosa.load(librosa.ex('trumpet'))

plt.figure(figsize=(16, 12))

# --- 1. 파형 (Waveform) ---
plt.subplot(2, 2, 1)
librosa.display.waveshow(y, sr=sr)
plt.title('Waveform (Time Domain)')
plt.xlabel('Time (s)')
plt.ylabel('Amplitude')

# --- 2. 스펙트로그램 (Spectrogram) ---
plt.subplot(2, 2, 2)
# STFT 수행 및 dB 변환
S = np.abs(librosa.stft(y))
D = librosa.amplitude_to_db(S, ref=np.max)
librosa.display.specshow(D, sr=sr, x_axis='time', y_axis='linear')
plt.title('Spectrogram (Hz Scale)')
plt.colorbar(format='%+2.0f dB')

# --- 3. 멜 스펙트로그램 (Mel-Spectrogram) ---
plt.subplot(2, 2, 3)
# 멜 스펙트로그램 계산
M = librosa.feature.melspectrogram(y=y, sr=sr, n_mels=128)
M_db = librosa.power_to_db(M, ref=np.max)
librosa.display.specshow(M_db, sr=sr, x_axis='time', y_axis='mel')
plt.title('Mel-Spectrogram (Mel Scale)')
plt.colorbar(format='%+2.0f dB')

# --- 4. MFCC ---
plt.subplot(2, 2, 4)
# MFCC 추출 (이미지의 설명대로 보통 13~20개 사용)
mfccs = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
librosa.display.specshow(mfccs, sr=sr, x_axis='time')
plt.title('MFCC (13 Coefficients)')
plt.colorbar()

plt.tight_layout()
plt.show()

```
