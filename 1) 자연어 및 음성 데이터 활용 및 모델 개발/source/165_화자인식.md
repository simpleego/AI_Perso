

### 2. 음성 특징 추출 실습 코드
주요 특징들(F0, RMS 에너지, 스펙트럼 특징)을 직접 추출하고 시각화하는 코드입니다. `librosa` 라이브러리를 사용하여 구현합니다.

```python
import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt

# 1. 음성 파일 로드 (샘플 데이터 사용)
y, sr = librosa.load(librosa.ex('choice'))

def extract_speech_features(y, sr):
    print(f"--- 주요 음성 특징 추출 ---")

    # [1] F0 (Fundamental Frequency) 추출 - 감정 분석의 핵심
    f0, voiced_flag, voiced_probs = librosa.pyin(y, fmin=librosa.note_to_hz('C2'), fmax=librosa.note_to_hz('C7'))
    avg_f0 = np.nanmean(f0)
    print(f"평균 기초 주파수(F0): {avg_f0:.2f} Hz")

    # [2] RMS Energy 추출 - 발화 강도 및 에너지 분석
    rms = librosa.feature.rms(y=y)
    avg_rms = np.mean(rms)
    print(f"평균 RMS 에너지: {avg_rms:.4f}")

    # [3] 스펙트럼 특징 (MFCC) - 화자 인식 및 현대적 접근의 기초
    # 이미지 2의 x-vector 등에 활용되는 저수준 특징량입니다.
    mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
    
    # --- 시각화 ---
    plt.figure(figsize=(12, 10))

    # Waveform & F0
    plt.subplot(3, 1, 1)
    librosa.display.waveshow(y, sr=sr, alpha=0.5)
    times = librosa.times_like(f0)
    plt.plot(times, f0, label='F0 (Pitch)', color='r', linewidth=2)
    plt.title('Waveform and Fundamental Frequency (F0)')
    plt.ylabel('Hz')
    plt.legend()

    # RMS Energy
    plt.subplot(3, 1, 2)
    plt.plot(librosa.times_like(rms), rms[0], label='RMS Energy', color='g')
    plt.title('Root Mean Square Energy')
    plt.ylabel('Amplitude')
    plt.legend()

    # MFCC Spectrogram (화자 특징 패턴)
    plt.subplot(3, 1, 3)
    librosa.display.specshow(mfcc, x_axis='time', sr=sr)
    plt.colorbar()
    plt.title('MFCC (Speaker Identity Features)')
    plt.ylabel('MFCC Coefficients')

    plt.tight_layout()
    plt.show()

# 함수 실행
extract_speech_features(y, sr)

```
