```python
import numpy as np
import matplotlib.pyplot as plt
import soundfile as sf  # Optional: file load/save

# ---------------------------------------------------------
# 1. 예시 오디오 신호 생성 (또는 실제 파일 로드)
# ---------------------------------------------------------
sample_rate = 44100
duration = 4.0  # 초
t = np.linspace(0, duration, int(sample_rate * duration))

# 작은 소리와 큰 소리가 번갈아 나타나는 신호 (예시)
# small amplitude (0.1) -> large (0.8) -> small -> large
signal = np.sin(2 * np.pi * 440 * t) * np.concatenate([
    0.1 * np.ones(int(t.size / 4)),
    0.8 * np.ones(int(t.size / 4)),
    0.1 * np.ones(int(t.size / 4)),
    0.8 * np.ones(int(t.size / 4))
])

# 실제 파일을 로드하려면 다음 코드를 사용하세요 (단, 이미지의 수식이 작동하려면 신호가 -1~1 범위여야 함)
# signal, sample_rate = sf.read('input_audio.wav')
# # 스테레오라면 모노로 변환
# if signal.ndim > 1: signal = signal[:, 0]
# signal = signal / np.max(np.abs(signal)) # Normalize to -1 ~ 1

# ---------------------------------------------------------
# 2. DRC (Dynamic Range Compression) 매개변수 설정
# ---------------------------------------------------------
# Threshold는 dB 스케일로 설정하는 것이 직관적입니다.
threshold_db = -18.0  # (dB)
ratio = 4.0          # (X:1)
makeup_gain_db = 6.0 # (dB)

# (옵션) Attack/Release는 복잡성을 줄이기 위해 생략합니다.

# ---------------------------------------------------------
# 3. DRC 알고리즘 구현
# ---------------------------------------------------------
# Step A: 입력 신호를 데시벨(dB) 스케일로 변환
# 0 값을 처리하기 위해 epsilon을 더해줍니다.
epsilon = 1e-12
input_signal_db = 20 * np.log10(np.abs(signal) + epsilon)

# Step B: 각 샘플에 대한 Gain Reduction(dB) 계산 (중심 수식)
gain_reduction_db = np.zeros_like(input_signal_db)

# 이미지의 수식 G(x)를 dB 스케일로 변환한 핵심 로직
over_threshold_indices = input_signal_db > threshold_db
if np.any(over_threshold_indices):
    over_by = input_signal_db[over_threshold_indices] - threshold_db
    # 실제 압축되는 양 = over_by * (1 - 1/Ratio)
    gain_reduction_db[over_threshold_indices] = -over_by * (1 - 1/ratio)

# Step C: Makeup Gain 적용 및 최종 dB Gain 계산
total_gain_db = gain_reduction_db + makeup_gain_db

# Step D: 출력 신호를 다시 리니어 스케일로 변환
output_signal_lin = np.multiply(signal, np.power(10, total_gain_db / 20))

# ---------------------------------------------------------
# 4. 결과 시각화
# ---------------------------------------------------------
plt.figure(figsize=(14, 10))

# 입력 및 출력 신호 파형
plt.subplot(3, 1, 1)
plt.plot(t, signal, label='Input Signal (Normalized)', alpha=0.7)
plt.plot(t, output_signal_lin, label='DRC + Makeup Gain Output', alpha=0.7, color='green')
plt.title('Audio Waveform: Input vs DRC Output')
plt.ylabel('Amplitude')
plt.legend()
plt.grid(True)

# 입력 및 출력 신호 데시벨(dB) 수준
plt.subplot(3, 1, 2)
# 임계값을 시각적으로 표시하기 위해 input_db를 plot
plt.plot(t, input_signal_db, label='Input Level (dB)', alpha=0.5)
plt.plot(t, 20 * np.log10(np.abs(output_signal_lin) + epsilon), label='Output Level (dB)', alpha=0.7, color='green')
plt.axhline(y=threshold_db, color='r', linestyle='--', label='Threshold')
plt.axhline(y=threshold_db - 6, color='y', linestyle='-', label='Compression Zone')
plt.title('Signal Level (dB): Threshold and Output')
plt.ylabel('Level (dB)')
plt.ylim([-40, 5])
plt.legend()
plt.grid(True)

# 적용된 Gain Reduction (dB)
plt.subplot(3, 1, 3)
plt.plot(t, total_gain_db, label='Total Gain (dB)', color='orange')
plt.plot(t, gain_reduction_db, label='Gain Reduction only (dB)', color='blue', alpha=0.5)
plt.title('Gain Reduction & Total Gain (dB)')
plt.ylabel('Gain (dB)')
plt.grid(True)
plt.legend()

plt.tight_layout()
plt.show()
```
