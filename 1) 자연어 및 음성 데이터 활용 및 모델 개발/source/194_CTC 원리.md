import torch
import torch.nn as nn
import torch.nn.functional as F

# 1. 간단한 CTC 기반 음성 인식 모델 구조 (RNN 활용)
class CTCModel(nn.Module):
    def __init__(self, input_dim, hidden_dim, num_classes):
        super(CTCModel, self).__init__()
        self.lstm = nn.LSTM(input_dim, hidden_dim, batch_first=True, bidirectional=True)
        # 출력 차원: 클래스 개수 + 1 (Blank 토큰 포함)
        self.fc = nn.Linear(hidden_dim * 2, num_classes + 1)

    def forward(self, x):
        # x: [Batch, Time, Feature]
        x, _ = self.lstm(x)
        logits = self.fc(x)
        # CTC 손실 함수는 log_softmax 형태의 확률값을 입력으로 받음
        return F.log_softmax(logits, dim=2)

# 2. 데이터 및 매개변수 설정
batch_size = 1
input_dim = 80       # 예: 80차원 Mel-spectrogram
hidden_dim = 128
num_classes = 26     # 예: 알파벳 26자

# 가상 데이터 생성 (T=100 프레임 입력, U=10 라벨 출력)
inputs = torch.randn(batch_size, 100, input_dim)
targets = torch.randint(low=1, high=num_classes, size=(batch_size, 10), dtype=torch.long)

# 입력 및 타겟의 실제 길이 정보 (CTC 계산에 필수)
input_lengths = torch.full(size=(batch_size,), fill_value=100, dtype=torch.long)
target_lengths = torch.full(size=(batch_size,), fill_value=10, dtype=torch.long)

# 3. 손실 함수 및 모델 선언
model = CTCModel(input_dim, hidden_dim, num_classes)
# blank 인덱스를 0으로 설정
ctc_loss = nn.CTCLoss(blank=0)

# 4. 순전파 및 손실 계산
# log_probs 차원: [Time, Batch, Class] (PyTorch 기본 사양)
log_probs = model(inputs).transpose(0, 1)

loss = ctc_loss(log_probs, targets, input_lengths, target_lengths)

print(f"CTC Loss: {loss.item():.4f}")
loss.backward()
print("역전파 완료: 정렬 정보를 수동으로 입력하지 않아도 모델이 모든 경로를 고려하여 학습합니다.")
