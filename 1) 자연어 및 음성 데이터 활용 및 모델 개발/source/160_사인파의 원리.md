# 사인파의 원리

> 사인파($Sine \ Wave$)가 생성되는 원리는 단위 원(Unit Circle) 위를 회전하는 점의 수직 높이($y$축 값)를 시간에 따라 나열하는 것입니다.   
> 점이 원을 한 바퀴($360^{\circ}$ 또는 $2\pi$ 라디안) 돌 때마다 사인파는 한 주기를 완성하게 됩니다.

이를 파이썬의 `matplotlib` 애니메이션 기능을 통해 시각화하면 원의 회전과 파형의 생성을 직관적으로 이해할 수 있습니다.

### 파이썬 시각화 코드 (애니메이션)

이 코드는 왼쪽에는 회전하는 단위 원을, 오른쪽에는 그에 따라 그려지는 사인파를 실시간으로 보여줍니다.

```python
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.animation import FuncAnimation

# 그래프 설정
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))

# 왼쪽: 단위 원 설정
theta = np.linspace(0, 2*np.pi, 100)
ax1.plot(np.cos(theta), np.sin(theta), color='gray', linestyle='--', alpha=0.5) # 가이드 원
ax1.set_xlim(-1.5, 1.5)
ax1.set_ylim(-1.5, 1.5)
ax1.set_aspect('equal')
ax1.set_title("Unit Circle (Rotation)")
ax1.grid(True)

point, = ax1.plot([], [], 'ro') # 회전하는 점
line_radius, = ax1.plot([], [], 'r-') # 중심에서 점까지의 선
line_y, = ax1.plot([], [], 'b-', linewidth=2) # y축 높이 표시

# 오른쪽: 사인파 설정
ax2.set_xlim(0, 2*np.pi)
ax2.set_ylim(-1.5, 1.5)
ax2.set_title("Sine Wave Generation")
ax2.set_xlabel("Phase (Radian)")
ax2.set_ylabel("Amplitude")
ax2.grid(True)

sine_line, = ax2.plot([], [], 'b-', linewidth=2)
trace_point, = ax2.plot([], [], 'ro')

# 애니메이션 데이터 초기화
x_data, y_data = [], []

def init():
    point.set_data([], [])
    line_radius.set_data([], [])
    line_y.set_data([], [])
    sine_line.set_data([], [])
    trace_point.set_data([], [])
    return point, line_radius, line_y, sine_line, trace_point

def update(frame):
    # 현재 각도(라디안)
    curr_theta = frame
    x = np.cos(curr_theta)
    y = np.sin(curr_theta)
    
    # 단위 원 업데이트
    point.set_data([x], [y])
    line_radius.set_data([0, x], [0, y])
    line_y.set_data([x, x], [0, y]) # 수직 높이(sin값) 강조
    
    # 사인파 업데이트
    x_data.append(curr_theta)
    y_data.append(y)
    sine_line.set_data(x_data, y_data)
    trace_point.set_data([curr_theta], [y])
    
    return point, line_radius, line_y, sine_line, trace_point

# 애니메이션 실행 (0부터 2pi까지 100단계)
frames = np.linspace(0, 2*np.pi, 100)
ani = FuncAnimation(fig, update, frames=frames, init_func=init, blit=True, interval=50)

plt.tight_layout()
plt.show()

```

---

### 💡 핵심 원리 설명

1. **회전 운동과 각도**: 원 위를 움직이는 점의 위치는 중심각 $\theta$에 의해 결정됩니다. 이때 수직 높이는 $\sin(\theta)$가 됩니다.
2. **시간의 흐름**: 점이 일정한 속도로 원을 돌 때, 이 시간을 $x$축으로 두고 높이($y$값)를 기록하면 우리가 아는 파형이 나타납니다.
3. **주기(Period)**: 점이 원을 한 바퀴($2\pi \approx 6.28$) 돌면 파형은 다시 원래 위치로 돌아와 다음 파동을 시작합니다.
4. **진폭(Amplitude)**: 원의 반지름이 커지면 파동의 높낮이(진폭)도 커집니다. 위 코드에서는 반지름이 **1**인 단위 원을 사용했습니다.

이 시각화는  "왜 사인파가 부드러운 곡선을 그리는가?"라는 질문에 대해, 원운동의 수직 성분을 펼쳐 놓은 것이라는 물리적 해답을 줄 수 있는 좋은 자료가 됩니다.

### (코랩/Jupyter 전용)

이 코드는 애니메이션을 `HTML5` 비디오로 변환하여 출력하므로, 중간에 끊기지 않고 전체 사인파가 생성되는 과정을 볼 수 있습니다.

```python
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.animation import FuncAnimation
from IPython.display import HTML

# 1. 그래프 및 데이터 초기화
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
x_data, y_data = [], [] # 데이터를 누적할 리스트

# 왼쪽: 단위 원
theta_circle = np.linspace(0, 2*np.pi, 100)
ax1.plot(np.cos(theta_circle), np.sin(theta_circle), color='gray', linestyle='--', alpha=0.5)
ax1.set_xlim(-1.5, 1.5)
ax1.set_ylim(-1.5, 1.5)
ax1.set_aspect('equal')
ax1.set_title("Unit Circle (Input)")

point, = ax1.plot([], [], 'ro') 
line_radius, = ax1.plot([], [], 'r-')

# 오른쪽: 사인파
ax2.set_xlim(0, 2*np.pi)
ax2.set_ylim(-1.2, 1.2)
ax2.set_title("Sine Wave (Output)")
sine_line, = ax2.plot([], [], 'b-', linewidth=2) # 실제로 그려질 선
trace_point, = ax2.plot([], [], 'ro')

def init():
    # 빈 데이터를 명시적으로 설정
    sine_line.set_data([], [])
    return sine_line,

def update(frame):
    # 2. 데이터 누적 (이 부분이 핵심입니다)
    x = np.cos(frame)
    y = np.sin(frame)
    
    x_data.append(frame)
    y_data.append(y)
    
    # 단위 원 업데이트
    point.set_data([x], [y])
    line_radius.set_data([0, x], [0, y])
    
    # 사인파 업데이트 (누적된 전체 데이터를 전달)
    sine_line.set_data(x_data, y_data)
    trace_point.set_data([frame], [y])
    
    return point, line_radius, sine_line, trace_point

# 3. 애니메이션 생성
frames = np.linspace(0, 2*np.pi, 60)
ani = FuncAnimation(fig, update, frames=frames, init_func=init, blit=True, interval=50)

# 4. 코랩에서 결과를 보기 위한 변환 (가장 중요)
plt.close() # 불필요한 빈 캔버스 닫기
HTML(ani.to_html5_video())

```
