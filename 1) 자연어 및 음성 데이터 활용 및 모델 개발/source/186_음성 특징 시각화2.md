# 음성 특징 시각화 실습

```python
import numpy as np
import matplotlib.pyplot as plt
import librosa
import librosa.display

# --------- 설정값 (강의에서 설명 포인트) ---------
AUDIO_PATH = "your_audio.wav"  # 실습용 음성 파일 경로
SR_TARGET = 16_000            # 리샘플링 샘플레이트 (음성 작업에서 자주 사용)
N_FFT      = 1024             # STFT 윈도우 크기 (≈ 64 ms @16kHz)
HOP_LENGTH = 256              # 프레임 간격 (≈ 16 ms @16kHz)
WIN_LENGTH = 1024             # 윈도우 길이
N_MELS     = 80               # 멜 필터 개수 (딥러닝 기본기)
N_MFCC     = 13               # MFCC 차원 (전통 특징 기본기)
# ------------------------------------------

# (선택) 프리엠퍼시스: 고주파 강조 y_pre[n] = y[n] - α y[n-1]
def pre_emphasis(y, alpha=0.97):
    y_emph = np.append(y[0], y[1:] - alpha * y[:-1])
    return y_emph

# 로드 & (선택) 프리엠퍼시스 적용
# 실제 파일이 없을 경우를 대비해 예외 처리를 하거나 librosa 내장 샘플을 사용할 수 있습니다.
try:
    y, sr = librosa.load(AUDIO_PATH, sr=SR_TARGET)
except:
    # 파일이 없을 경우 트럼펫 샘플로 대체 (테스트용)
    y, sr = librosa.load(librosa.ex('trumpet'), sr=SR_TARGET)

y_emph = pre_emphasis(y, alpha=0.97)

# -------- 1) Waveform --------
plt.figure(figsize=(10, 3))
librosa.display.waveshow(y_emph, sr=sr)
plt.title("Waveform (pre-emphasized)")
plt.xlabel("Time (s)")
plt.ylabel("Amplitude")
plt.tight_layout()
plt.show()

# -------- 2) Spectrogram (STFT) --------
# STFT: X(m, ω) = Σ_n x[n] w[n-m] e^{-j ω n}
S_complex = librosa.stft(
    y_emph, n_fft=N_FFT, hop_length=HOP_LENGTH, win_length=WIN_LENGTH, window="hann"
)

S_mag = np.abs(S_complex) # |X|
S_db = librosa.amplitude_to_db(S_mag, ref=np.max)

plt.figure(figsize=(10, 4))
librosa.display.specshow(
    S_db, sr=sr, hop_length=HOP_LENGTH, x_axis="time", y_axis="linear"
)
plt.title("Spectrogram (|STFT| in dB)")
plt.colorbar(format="%+1.1f dB")
plt.tight_layout()
plt.show()

# -------- 3) Mel-Spectrogram --------
# Mel 변환: m = 2595 * log10(1 + f/700)
M = librosa.feature.melspectrogram(
    y=y_emph, sr=sr, n_fft=N_FFT, hop_length=HOP_LENGTH, n_mels=N_MELS, power=2.0
) # power=2.0 -> |X|^2
M_db = librosa.power_to_db(M, ref=np.max)

plt.figure(figsize=(10, 4))
librosa.display.specshow(
    M_db, sr=sr, hop_length=HOP_LENGTH, x_axis="time", y_axis="mel"
)
plt.title("Mel-Spectrogram (dB)")
plt.colorbar(format="%+1.1f dB")
plt.tight_layout()
plt.show()

# -------- 4) MFCC --------
# c_k = Σ_{m=1..M} log(S_m) * cos[π k / M * (m - 0.5)]
MFCC = librosa.feature.mfcc(S=librosa.power_to_db(M), n_mfcc=N_MFCC)

plt.figure(figsize=(10, 4))
librosa.display.specshow(
    MFCC, x_axis="time", sr=sr, hop_length=HOP_LENGTH
)
plt.title(f"MFCC (n_mfcc={N_MFCC})")
plt.ylabel("MFCC Coefficients")
plt.colorbar()
plt.tight_layout()
plt.show()
