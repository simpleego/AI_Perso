# 음성인식 파이프라인

```python
import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt

def speech_analysis_pipeline(file_path):
    # 1. 데이터 로드 및 무음 제거 (Preprocessing)
    # y: 신호, sr: 샘플링 레이트
    y_raw, sr = librosa.load(file_path)
    
    # top_db 기준 이하의 소음을 잘라냄 (무음 제거)
    y, index = librosa.effects.trim(y_raw, top_db=20)
    
    print(f"✅ 분석 시작: 원본 길이 {len(y_raw)/sr:.2f}s -> 트리밍 후 {len(y)/sr:.2f}s")

    # 2. 화자 및 음색 특징 (MFCC)
    # 화자의 고유한 성도(Vocal Tract) 특징을 반영
    mfccs = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
    
    # 3. 기초 주파수 (F0 - 감정 지표)
    # 성대 진동 횟수를 추정 (PYIN 알고리즘)
    f0, voiced_flag, voiced_probs = librosa.pyin(y, fmin=librosa.note_to_hz('C2'), fmax=librosa.note_to_hz('C7'))
    avg_f0 = np.nanmean(f0)

    # 4. 에너지 (Energy - 강조 및 발화 강도)
    rms = librosa.feature.rms(y=y)
    avg_rms = np.mean(rms)

    # 5. 스펙트럼 중심 (Spectral Centroid - 밝기 및 포먼트 대용)
    # 소리의 '밝기' 혹은 '날카로움'을 측정
    spec_centroid = librosa.feature.spectral_centroid(y=y, sr=sr)
    avg_centroid = np.mean(spec_centroid)

    # --- 종합 리포트 출력 ---
    print("\n" + "="*30)
    print("📊 음성 분석 종합 리포트")
    print("-" * 30)
    print(f"1. Pitch (F0)     : {avg_f0:.2f} Hz (감정/높낮이)")
    print(f"2. Energy (RMS)   : {avg_rms:.4f} (강세/세기)")
    print(f"3. Timbre (Centroid): {avg_centroid:.2f} Hz (음색 밝기)")
    print(f"4. Speaker Pattern: MFCC {mfccs.shape} 추출 완료")
    print("="*30)

    # --- 시각화 파이프라인 ---
    plt.figure(figsize=(14, 12))

    # (1) Waveform & F0 (Pitch Contour)
    plt.subplot(4, 1, 1)
    librosa.display.waveshow(y, sr=sr, alpha=0.4)
    plt.plot(librosa.times_like(f0), f0, label='Pitch (F0)', color='red', linewidth=2)
    plt.title('Waveform & Fundamental Frequency (F0)')
    plt.legend(loc='upper right')

    # (2) MFCC (Speaker Identity)
    plt.subplot(4, 1, 2)
    librosa.display.specshow(mfccs, x_axis='time', sr=sr)
    plt.colorbar()
    plt.title('MFCC (Speaker Identity Map)')

    # (3) Spectral Centroid (Brightness)
    plt.subplot(4, 1, 3)
    plt.semilogy(librosa.times_like(spec_centroid), spec_centroid[0], label='Spectral Centroid', color='orange')
    plt.title('Spectral Centroid (Voice Brightness)')
    plt.ylabel('Hz')
    plt.legend()

    # (4) RMS Energy
    plt.subplot(4, 1, 4)
    plt.plot(librosa.times_like(rms), rms[0], label='RMS Energy', color='green')
    plt.title('Short-term Energy (Loudness)')
    plt.xlabel('Time (s)')
    plt.legend()

    plt.tight_layout()
    plt.show()

# 실습 실행 (librosa 제공 샘플 활용)
sample_file = librosa.ex('choice')
speech_analysis_pipeline(sample_file)
```
