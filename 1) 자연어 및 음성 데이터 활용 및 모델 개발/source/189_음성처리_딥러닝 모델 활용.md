```python
import torch
import torch.nn as nn

# 입력 데이터 가정: (Batch, Time_Steps, Feature_Dim)
# 예: 8개의 데이터를 100프레임(시간), 80차원(멜스펙)으로 구성
input_data = torch.randn(8, 100, 80)

# 1. RNN (LSTM) 레이어
# 순차적으로 데이터를 훑으며 상태를 업데이트합니다.
lstm = nn.LSTM(input_size=80, hidden_size=128, num_layers=2, batch_first=True)
lstm_out, (hn, cn) = lstm(input_data)
print(f"LSTM 출력 크기: {lstm_out.shape}") # [8, 100, 128]

# 2. CNN 레이어
# 음성을 "그림"으로 보고 지역적 패턴을 찾습니다.
# CNN은 (Batch, Channel, Height, Width) 형식을 선호하므로 차원 변경이 필요합니다.
cnn_input = input_data.unsqueeze(1) # [8, 1, 100, 80] (채널 추가)
cnn = nn.Sequential(
    nn.Conv2d(1, 32, kernel_size=(3, 3), stride=(2, 2), padding=(1, 1)),
    nn.ReLU(),
    nn.MaxPool2d(2)
)
cnn_out = cnn(cnn_input)
print(f"CNN 출력 크기: {cnn_out.shape}")

# 3. Transformer (Self-Attention) 레이어
# 모든 시간대 사이의 관계를 한 번에 계산합니다.
# d_model은 입력 차원과 같아야 합니다.
transformer_layer = nn.TransformerEncoderLayer(d_model=80, nhead=8, batch_first=True)
transformer_out = transformer_layer(input_data)
print(f"Transformer 출력 크기: {transformer_out.shape}") # [8, 100, 80]

```
