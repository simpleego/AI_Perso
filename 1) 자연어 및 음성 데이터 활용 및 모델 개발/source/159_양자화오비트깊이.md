## 디지털화의 기초: 양자화와 비트값

### 1. 양자화 (Quantization)
- 연속적인 진폭 값을 유한한 레벨로 근사

* [샘플링 양자화 보기](https://mz-moonzoo.tistory.com/62)
* [샘플링 양자화 보기2](https://www.playsw.or.kr/artificial/textbook/detail/14)
 
### 2. 비트 값 (Bit Depth)
- 각 샘플 표현에 쓰이는 비트 수: **N_levels = 2^B**
- 8-bit → 256 레벨 (품질 낮음)
- 16-bit → 65,536 레벨 (표준)
- 24-bit → 16,777,216 레벨 (전문 녹음)

### 3. 동적 범위 (Dynamic Range)
- 신호가 표현할 수 있는 최대/최소 비율
- 근사식: **DR ≈ 6.02 × B (dB)**
- 예: 16-bit → 약 96 dB

### 4. 파일 포맷

| 포맷 | 특성 | 장점 | 단점 | 용도 |
|------|------|------|------|------|
| **WAV** | 무손실, PCM 기반 | 고품질, 편집/처리 적합 | 용량 큼 (115MB/시간) | 음성인식 연구, 데이터셋 |
| **MP3** | 손실 압축 | 용량 작음 (WAV 대비 1/10) | 압축 아티팩트, ASR 성능 저하 | 음악 스트리밍 |
| **AAC** | MP3 후속, 높은 압축 효율 | 동일 비트레이트에서 더 나은 음질 | - | 유튜브, 애플뮤직, 디지털 방송 |

# 실습 코드 (Python)

```python
import numpy as np
import matplotlib.pyplot as plt
import librosa
import librosa.display
import soundfile as sf
from scipy import signal

# 1. 비트 깊이에 따른 양자화 시각화
def visualize_quantization():
    plt.figure(figsize=(12, 8))
    
    # 원본 신호 (1초, 1kHz 샘플링)
    t = np.linspace(0, 1, 1000)
    original = np.sin(2 * np.pi * 5 * t)
    
    # 다양한 비트 깊이로 양자화
    bit_depths = [2, 3, 4, 8]
    
    for i, bits in enumerate(bit_depths):
        levels = 2 ** bits
        # 양자화 스텝
        step = 2 / levels
        # 양자화
        quantized = np.round(original / step) * step
        quantized = np.clip(quantized, -1, 1)
        
        plt.subplot(2, 2, i+1)
        plt.plot(t[:200], original[:200], 'b-', alpha=0.5, label='Original')
        plt.plot(t[:200], quantized[:200], 'r.', markersize=2, label=f'{bits}-bit ({levels} levels)')
        plt.title(f'Quantization - {bits} bit', fontsize=12)
        plt.xlabel('Time (s)')
        plt.ylabel('Amplitude')
        plt.legend()
        plt.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()

# 2. 비트 깊이에 따른 SNR 계산
def calculate_snr_by_bit_depth():
    print("=" * 50)
    print("비트 깊이에 따른 이론적 SNR (Signal-to-Noise Ratio)")
    print("=" * 50)
    
    for bits in [8, 16, 24]:
        dr_db = 6.02 * bits
        print(f"{bits}-bit: DR ≈ {dr_db:.1f} dB, 최대 SNR ≈ {dr_db:.1f} dB")
    
    print("\n※ SQNR = 6.02B + 1.76 dB (정현파 기준)")

# 3. WAV vs MP3 비교
def compare_audio_formats(audio_path=None):
    """
    오디오 파일 로드 및 포맷 비교
    audio_path가 없으면 예제 신호 생성
    """
    if audio_path:
        # WAV 파일 로드
        wav_data, sr = librosa.load(audio_path, sr=None)
        file_type = "WAV"
    else:
        # 테스트 신호 생성 (440Hz 사인파)
        duration = 3  # 3초
        sr = 16000
        t = np.linspace(0, duration, int(sr * duration))
        wav_data = np.sin(2 * np.pi * 440 * t)
        # 페이드 인/아웃
        fade_len = int(0.01 * sr)
        fade_in = np.linspace(0, 1, fade_len)
        fade_out = np.linspace(1, 0, fade_len)
        wav_data[:fade_len] *= fade_in
        wav_data[-fade_len:] *= fade_out
        file_type = "생성된 사인파"
    
    print("=" * 50)
    print("오디오 포맷 비교")
    print("=" * 50)
    print(f"파일 타입: {file_type}")
    print(f"샘플링 레이트: {sr} Hz")
    print(f"데이터 타입: {wav_data.dtype}")
    print(f"최대 진폭: {np.max(np.abs(wav_data)):.4f}")
    print(f"RMS 에너지: {np.sqrt(np.mean(wav_data**2)):.4f}")
    
    # 스펙트로그램 표시
    plt.figure(figsize=(12, 5))
    
    plt.subplot(1, 2, 1)
    plt.plot(wav_data[:sr//10])  # 0.1초만 표시
    plt.title('Waveform (0.1초)')
    plt.xlabel('Sample')
    plt.ylabel('Amplitude')
    
    plt.subplot(1, 2, 2)
    D = librosa.amplitude_to_db(np.abs(librosa.stft(wav_data)), ref=np.max)
    img = librosa.display.specshow(D, sr=sr, x_axis='time', y_axis='hz')
    plt.colorbar(img, format='%+2.0f dB')
    plt.title('Spectrogram')
    
    plt.tight_layout()
    plt.show()
    
    return wav_data, sr

# 4. 양자화 오차 분석
def quantization_error_analysis():
    """양자화 오차의 통계적 특성 분석"""
    np.random.seed(42)
    
    # 다양한 비트 깊이로 음성 신호 양자화 시뮬레이션
    duration = 0.1  # 0.1초
    sr = 16000
    t = np.linspace(0, duration, int(sr * duration))
    
    # 음성 유사 신호 (주파수 변조)
    original = np.sin(2 * np.pi * 200 * t) * np.exp(-t * 10)
    original = original / np.max(np.abs(original))
    
    plt.figure(figsize=(14, 10))
    
    bit_depths = [4, 8, 16]
    
    for idx, bits in enumerate(bit_depths):
        levels = 2 ** bits
        step = 2 / levels
        
        # 양자화
        quantized = np.round(original / step) * step
        quantized = np.clip(quantized, -1, 1)
        
        # 양자화 오차
        error = quantized - original
        
        # 오차 통계
        print(f"\n{bits}-bit 양자화 오차 통계:")
        print(f"  평균 오차: {np.mean(error):.6f}")
        print(f"  오차 분산: {np.var(error):.6f}")
        print(f"  최대 오차: {np.max(np.abs(error)):.6f}")
        print(f"  RMSE: {np.sqrt(np.mean(error**2)):.6f}")
        
        plt.subplot(3, 3, idx*3 + 1)
        plt.plot(t[:200], original[:200], 'b-', alpha=0.6, label='Original')
        plt.plot(t[:200], quantized[:200], 'r-', alpha=0.6, label=f'{bits}-bit')
        plt.title(f'{bits}-bit Signal')
        plt.legend()
        
        plt.subplot(3, 3, idx*3 + 2)
        plt.hist(error, bins=50, edgecolor='black')
        plt.title(f'{bits}-bit Quantization Error Histogram')
        plt.xlabel('Error')
        plt.ylabel('Frequency')
        
        plt.subplot(3, 3, idx*3 + 3)
        plt.plot(t[:200], error[:200])
        plt.title(f'{bits}-bit Error (Time Domain)')
        plt.xlabel('Sample')
        plt.ylabel('Error')
    
    plt.tight_layout()
    plt.show()

# 5. WAV 파일 저장 및 정보 확인
def save_and_analyze_wav(data, sr, filename="output.wav"):
    """WAV 파일 저장 및 상세 정보 출력"""
    # 데이터 정규화 (int16 범위)
    data_int16 = (data * 32767).astype(np.int16)
    
    # 파일 저장
    sf.write(filename, data, sr, subtype='PCM_16')
    
    # 파일 정보
    import os
    file_size = os.path.getsize(filename)
    
    print("=" * 50)
    print("WAV 파일 정보")
    print("=" * 50)
    print(f"파일명: {filename}")
    print(f"파일 크기: {file_size:,} bytes ({file_size/1024:.1f} KB)")
    print(f"샘플링 레이트: {sr} Hz")
    print(f"비트 깊이: 16-bit")
    print(f"채널 수: 1 (Mono)")
    print(f"예상 용량: {file_size/1024:.1f} KB")
    print(f"실시간 재생 시간: {len(data)/sr:.2f}초")
    
    # 시간당 예상 용량
    bytes_per_second = sr * 2  # 16-bit = 2 bytes
    print(f"\n용량 추정:")
    print(f"  초당: {bytes_per_second:,} bytes ({bytes_per_second/1024:.1f} KB/s)")
    print(f"  분당: {bytes_per_second * 60 / 1024:.1f} KB/분")
    print(f"  시간당: {bytes_per_second * 3600 / 1024 / 1024:.1f} MB/시간")

# 6. MP3 변환 및 품질 비교 (pydub 필요)
def demo_mp3_conversion():
    """
    MP3 변환 데모 (pydub 및 ffmpeg 필요)
    pip install pydub
    """
    try:
        from pydub import AudioSegment
        import os
        
        # 테스트 WAV 생성
        sr = 16000
        duration = 2
        t = np.linspace(0, duration, int(sr * duration))
        # 더 복잡한 신호 (화음)
        data = (0.5 * np.sin(2 * np.pi * 440 * t) + 
                0.3 * np.sin(2 * np.pi * 880 * t) +
                0.2 * np.sin(2 * np.pi * 1320 * t))
        data = data / np.max(np.abs(data))
        
        # WAV 저장
        sf.write("temp.wav", data, sr)
        
        # MP3 변환 (다양한 비트레이트)
        audio = AudioSegment.from_wav("temp.wav")
        
        print("=" * 50)
        print("MP3 압축 비교")
        print("=" * 50)
        
        for bitrate in [64, 128, 192, 320]:
            mp3_file = f"temp_{bitrate}k.mp3"
            audio.export(mp3_file, format="mp3", bitrate=f"{bitrate}k")
            
            # 파일 크기 비교
            wav_size = os.path.getsize("temp.wav")
            mp3_size = os.path.getsize(mp3_file)
            
            print(f"\n{bitrate}kbps MP3:")
            print(f"  크기: {mp3_size:,} bytes ({mp3_size/wav_size*100:.1f}% of WAV)")
            print(f"  압축률: {wav_size/mp3_size:.1f}x")
        
        # 임시 파일 정리
        os.remove("temp.wav")
        for bitrate in [64, 128, 192, 320]:
            os.remove(f"temp_{bitrate}k.mp3")
            
    except ImportError:
        print("pydub가 설치되지 않았습니다. (pip install pydub 필요)")
    except Exception as e:
        print(f"MP3 변환 중 오류: {e}")

# 메인 실행
if __name__ == "__main__":
    print("🎵 오디오 디지털화 기초 실습 🎵\n")
    
    # 1. 양자화 시각화
    print("1. 비트 깊이별 양자화 시각화...")
    visualize_quantization()
    
    # 2. SNR 계산
    print("\n2. 이론적 SNR 계산...")
    calculate_snr_by_bit_depth()
    
    # 3. 오디오 포맷 비교
    print("\n3. 오디오 포맷 분석...")
    data, sr = compare_audio_formats()
    
    # 4. 양자화 오차 분석
    print("\n4. 양자화 오차 분석...")
    quantization_error_analysis()
    
    # 5. WAV 파일 저장
    print("\n5. WAV 파일 저장...")
    save_and_analyze_wav(data, sr)
    
    # 6. MP3 변환 데모 (선택사항)
    print("\n6. MP3 압축 비교...")
    demo_mp3_conversion()
    
    print("\n✅ 실습 완료!")
```
