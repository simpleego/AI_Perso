
```python
import whisper # pip install openai-whisper

# 1. 모델 로드 (크기에 따라 tiny, base, small, medium, large 선택 가능)
# 'base' 모델은 속도와 정확도의 밸런스가 좋습니다.
model = whisper.load_model("base")

# 2. 음성 파일 경로 설정
# (직접 녹음한 파일이나 제공된 실습용 wav 파일을 사용하세요)
AUDIO_PATH = "speech_sample.wav"

# 3. 음성 인식 수행 (Transcribe)
# fp16=False는 CPU 환경에서 실행 시 에러 방지를 위해 설정합니다.
result = model.transcribe(AUDIO_PATH, fp16=False, language="ko")

# 4. 결과 출력
print("-" * 30)
print(f"인식 결과: {result['text']}")
print("-" * 30)

# 상세 타임스탬프 확인 (회의록 작성 등에 활용)
for segment in result['segments']:
    print(f"[{segment['start']:.2f}s -> {segment['end']:.2f}s]: {segment['text']}")

```
