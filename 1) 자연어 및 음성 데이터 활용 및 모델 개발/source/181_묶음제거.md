```python
import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt

# 1. 예시 오디오 로드 (또는 테스트용 가상 신호 생성)
# librosa의 내장 예제 데이터 로드 (클린한 음성)
path = librosa.ex('trumpet')
y, sr = librosa.load(path)

# 2. 묵음 제거 (Trim)
# top_db: 임계값 (최대 데시벨 대비 낮게 설정된 값 이하를 묵음 처리)
# 예: 20db로 설정하면 최고 음량보다 20db 낮은 소리들은 묵음으로 간주
y_trimmed, index = librosa.effects.trim(y, top_db=20)

print(f"원본 신호 길이: {len(y)/sr:.2f}초")
print(f"제거 후 신호 길이: {len(y_trimmed)/sr:.2f}초")
print(f"제거된 구간 (샘플 단위): {index}")

# 3. 시각화 비교
plt.figure(figsize=(12, 6))

plt.subplot(2, 1, 1)
librosa.display.waveshow(y, sr=sr, alpha=0.5)
plt.title('Original Signal (with Silence)')
plt.vlines(index/sr, -1, 1, color='r', linestyle='--', label='Trim Points')
plt.legend()

plt.subplot(2, 1, 2)
librosa.display.waveshow(y_trimmed, sr=sr, color='g')
plt.title('Trimmed Signal (Silence Removed from Ends)')

plt.tight_layout()
plt.show()
```
