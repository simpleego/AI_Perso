```python
import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt

# 1. 오디오 로드
y, sr = librosa.load(librosa.ex('choice'))

# 2. 특징 추출
# A. STFT -> Magnitude Spectrogram
stft = np.abs(librosa.stft(y))
db_stft = librosa.amplitude_to_db(stft, ref=np.max)

# B. Mel-Spectrogram
mel_spec = librosa.feature.melspectrogram(y=y, sr=sr, n_mels=128)
db_mel = librosa.power_to_db(mel_spec, ref=np.max)

# C. MFCC
mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=20)

# 3. 시각화
plt.figure(figsize=(15, 12))

# Spectrogram (STFT)
plt.subplot(3, 1, 1)
librosa.display.specshow(db_stft, sr=sr, x_axis='time', y_axis='hz')
plt.title('STFT (Spectrogram in Hz scale)')
plt.colorbar(format='%+2.0f dB')

# Mel-Spectrogram
plt.subplot(3, 1, 2)
librosa.display.specshow(db_mel, sr=sr, x_axis='time', y_axis='mel')
plt.title('Mel-Spectrogram (Mel scale)')
plt.colorbar(format='%+2.0f dB')

# MFCC
plt.subplot(3, 1, 3)
librosa.display.specshow(mfcc, x_axis='time')
plt.title('MFCC (Cepstral Coefficients)')
plt.colorbar()

plt.tight_layout()
plt.show()

```
