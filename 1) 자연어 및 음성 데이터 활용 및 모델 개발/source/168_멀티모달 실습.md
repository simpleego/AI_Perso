
### 🎥 음성 + 영상 결합 분석 실습: "시청각 동기화 특징 추출"

이 코드는 영상의 시각적 변화량(움직임)과 음성의 에너지(소리 크기)를 시간에 따라 매핑하여 시각화합니다.

```python
import cv2
import librosa
import numpy as np
import matplotlib.pyplot as plt

def analyze_audio_video_sync(video_path):
    # 1. 영상 처리 (Video Processing) - 프레임별 밝기 변화 추출
    cap = cv2.VideoCapture(video_path)
    fps = cap.get(cv2.CAP_PROP_FPS)
    frame_brightness = []

    print("📹 영상을 분석 중입니다...")
    while cap.isOpened():
        ret, frame = cap.read()
        if not ret:
            break
        # 그레이스케일 변환 후 평균 밝기 계산 (움직임이나 화면 전환 감지)
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        frame_brightness.append(np.mean(gray))
    cap.release()

    # 2. 음성 처리 (Audio Processing) - 에너지 추출
    # 영상 파일에서 음성을 직접 로드 (librosa는 영상 파일 내 오디오 트랙 읽기 지원)
    y, sr = librosa.load(video_path)
    rms = librosa.feature.rms(y=y)[0]
    
    # 음성 시간축 생성
    audio_times = librosa.times_like(rms, sr=sr)
    # 영상 시간축 생성 (프레임 인덱스를 시간으로 변환)
    video_times = np.arange(len(frame_brightness)) / fps

    # 3. 데이터 정규화 (비교를 위해 0~1 사이값으로 조정)
    norm_brightness = (frame_brightness - np.min(frame_brightness)) / (np.max(frame_brightness) - np.min(frame_brightness))
    norm_rms = (rms - np.min(rms)) / (np.max(rms) - np.min(rms))

    # 4. 통합 시각화 리포트
    plt.figure(figsize=(12, 8))

    # 상단: 영상 밝기 변화 (Visual)
    plt.subplot(2, 1, 1)
    plt.plot(video_times, norm_brightness, color='blue', label='Visual Brightness (normalized)')
    plt.title('Audio-Visual Synchronization Analysis')
    plt.ylabel('Visual Intensity')
    plt.legend()

    # 하단: 음성 에너지 변화 (Audio)
    plt.subplot(2, 1, 2)
    plt.plot(audio_times, norm_rms, color='red', label='Audio Energy (RMS normalized)')
    plt.ylabel('Audio Intensity')
    plt.xlabel('Time (seconds)')
    plt.legend()

    plt.tight_layout()
    plt.show()

# 실제 실행 시에는 영상 파일 경로(.mp4 등)를 입력하세요.
# analyze_audio_video_sync('sample_video.mp4')
print("💡 실행 팁: .mp4 파일을 업로드한 뒤 함수를 호출하면 영상과 음성의 싱크를 그래프로 볼 수 있습니다.")

```
