import numpy as np
import sounddevice as sd

# 1. 기본 설정
sr = 16000      # 샘플링 레이트 (1초에 16000개 샘플)
duration = 0.5  # 각 음의 길이 (0.5초)
t = np.linspace(0, duration, int(sr * duration), endpoint=False) # 시간축 생성

# 2. 음계별 주파수 정의 (4옥타브 기준 Hz)
notes = {
    '도': 261.63,
    '레': 293.66,
    '미': 329.63,
    '파': 349.23,
    '솔': 392.00,
    '라': 440.00,
    '시': 493.88,
    '높은도': 523.25
}

def generate_tone(frequency, sr, t):
    """특정 주파수의 사인파를 생성하고 소리를 부드럽게 만들기 위해 페이드 인/아웃 적용"""
    # 기본 사인파 생성
    wave = np.sin(2 * np.pi * frequency * t)
    
    # 뚝 끊기는 소리(클릭 노이즈)를 방지하기 위한 간단한 외포선(Envelope) 적용
    envelope = np.ones_like(t)
    fade_len = int(sr * 0.05) # 0.05초 동안 페이드
    envelope[:fade_len] = np.linspace(0, 1, fade_len)
    envelope[-fade_len:] = np.linspace(1, 0, fade_len)
    
    return wave * envelope

# 3. 사용자 입력 및 출력 부분
print("--- 간단한 파이썬 피아노 ---")
print("입력 가능: 도, 레, 미, 파, 솔, 라, 시, 높은도 (종료: q)")

while True:
    user_input = input("\n연주할 음을 입력하세요: ").strip()
    
    if user_input.lower() == 'q':
        print("프로그램을 종료합니다.")
        break
        
    if user_input in notes:
        freq = notes[user_input]
        wave = generate_tone(freq, sr, t)
        
        print(f"🎵 '{user_input}' ({freq} Hz) 재생 중...")
        sd.play(wave, sr)
        sd.wait()  # 재생이 끝날 때까지 대기
    else:
        print("❌ '도, 레, 미...' 형식으로 정확히 입력해주세요.")

## 학교종이 땡땡땡 연주

```python
import numpy as np
from IPython.display import Audio, display
import time

# 1. 기본 설정
sr = 16000
duration = 0.4  # 기본 음 길이 (0.4초)
t = np.linspace(0, duration, int(sr * duration), endpoint=False)

# 2. 음계별 주파수 정의
notes = {
    '도': 261.63, '레': 293.66, '미': 329.63, '파': 349.23,
    '솔': 392.00, '라': 440.00, '시': 493.88, '높은도': 523.25,
    ' ': 0  # 무음(쉼표) 처리용
}

def generate_tone(frequency, sr, t):
    if frequency == 0:
        return np.zeros_like(t)

    wave = np.sin(2 * np.pi * frequency * t)
    # 부드러운 연결을 위한 엔벨로프
    envelope = np.ones_like(t)
    fade_len = int(sr * 0.05)
    envelope[:fade_len] = np.linspace(0, 1, fade_len)
    envelope[-fade_len:] = np.linspace(1, 0, fade_len)
    return wave * envelope

def play_melody(melody_list):
    """음계 리스트를 받아 하나의 오디오로 합쳐 재생"""
    combined_wave = []

    for note in melody_list:
        if note in notes:
            freq = notes[note]
            wave = generate_tone(freq, sr, t)
            combined_wave.append(wave)
        else:
            print(f"⚠️ 경고: '{note}'는 알 수 없는 음계입니다. 건너뜁니다.")

    if combined_wave:
        # 모든 음을 하나로 이어 붙임
        final_wave = np.concatenate(combined_wave)
        display(Audio(final_wave, rate=sr, autoplay=True))

# 3. 메인 실행부
# 샘플: 학교 종이 땡땡땡
school_bell = ['솔', '솔', '라', '라', '솔', '솔', '미', ' ',
               '솔', '솔', '미', '미', '레', ' ',
               '솔', '솔', '라', '라', '솔', '솔', '미', ' ',
               '솔', '미', '레', '미', '도']

print("🔔 샘플 연주: 학교 종이 땡땡땡")
play_melody(school_bell)

print("\n" + "="*40)
print("🎹 나만의 멜로디 연주하기")
print("입력 예시: 도 레 미 파 솔 라 시 높은도")
print("(음계 사이에 공백을 넣어주세요. 종료는 q)")
print("="*40)

while True:
    user_input = input("\n연주할 악보를 입력하세요: ").strip()

    if user_input.lower() == 'q':
        print("프로그램을 종료합니다.")
        break

    # 공백을 기준으로 리스트화
    user_melody = user_input.split()
    print(f"🎼 입력하신 멜로디를 연주합니다: {user_melody}")
    play_melody(user_melody)
```


# PC버전 피아노

```python
import numpy as np
import sounddevice as sd
from pynput import keyboard
import time

# 1. 설정
SR = 44100
CHANNELS = 1
BLOCK_SIZE = 512 # 버퍼 크기 (작을수록 반응 속도 빠름, 너무 작으면 뚝뚝 끊김)

# 음계별 주파수
NOTES = {
    'a': 261.63, 's': 293.66, 'd': 329.63, 'f': 349.23,
    'g': 392.00, 'h': 440.00, 'j': 493.88, 'k': 523.25
}

# 현재 재생 중인 음의 주파수와 위상을 관리
current_notes = {} # {key: {'freq': 261.63, 'phase': 0.0}}

def audio_callback(outdata, frames, time_info, status):
    """오디오 장치가 데이터를 요청할 때마다 호출되는 콜백 함수"""
    if status:
        print(status)
    
    # 출력 버퍼 초기화
    out_signal = np.zeros(frames)
    
    # 현재 눌려 있는 모든 음을 합성 (Mixer 역할)
    for key, info in list(current_notes.items()):
        freq = info['freq']
        phase = info['phase']
        
        # 주파수에 따른 사인파 생성 및 위상 관리
        t = (np.arange(frames) + phase) / SR
        out_signal += 0.3 * np.sin(2 * np.pi * freq * t) # 0.3은 볼륨 조절 (화음 시 깨짐 방지)
        
        # 다음 블록을 위한 위상 업데이트
        current_notes[key]['phase'] = (phase + frames)
        
    outdata[:] = out_signal.reshape(-1, 1)

def on_press(key):
    try:
        char = key.char
        if char in NOTES and char not in current_notes:
            # 새로운 음 추가 (위상은 0부터 시작)
            current_notes[char] = {'freq': NOTES[char], 'phase': 0.0}
    except AttributeError:
        pass

def on_release(key):
    try:
        char = key.char
        if char in current_notes:
            # 음 제거
            del current_notes[char]
    except AttributeError:
        pass
    if key == keyboard.Key.esc:
        return False

# 2. 메인 스트림 시작
print("🎹 [개선된 믹서 버전] 실시간 피아노")
print("A S D F G H J K 키를 누르세요. (ESC: 종료)")

# 단 하나의 스트림만 유지 (Callback 방식이 가장 안정적임)
with sd.OutputStream(samplerate=SR, channels=CHANNELS, callback=audio_callback, blocksize=BLOCK_SIZE):
    with keyboard.Listener(on_press=on_press, on_release=on_release) as listener:
        listener.join()
```
