
### 2. 심화 실습 코드: "환경 변화와 발화 길이의 영향"

이미지 3의 '기술적 과제'를 직접 실험해볼 수 있도록, **소음을 섞거나 음성 길이를 잘랐을 때 화자 식별 특징(MFCC)이 얼마나 변하는지** 확인하는 코드를 구성했습니다.

```python
import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt

# 1. 샘플 데이터 로드
y, sr = librosa.load(librosa.ex('choice'), duration=5)

def simulate_technical_challenges(y, sr):
    # --- [과제 1] 환경 변화 (백색 소음 추가) ---
    noise = np.random.randn(len(y))
    y_noisy = y + 0.02 * noise  # 약한 소음 추가
    
    # --- [과제 2] 발화 길이 (0.5초로 짧게 자르기) ---
    y_short = y[:int(sr * 0.5)]

    # 특징 추출 (MFCC)
    mfcc_orig = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
    mfcc_noisy = librosa.feature.mfcc(y=y_noisy, sr=sr, n_mfcc=13)
    mfcc_short = librosa.feature.mfcc(y=y_short, sr=sr, n_mfcc=13)

    # 시각화
    plt.figure(figsize=(15, 10))

    # 원본 MFCC
    plt.subplot(3, 1, 1)
    librosa.display.specshow(mfcc_orig, sr=sr, x_axis='time')
    plt.title('Original MFCC (Clean & Long)')
    plt.colorbar()

    # 소음 추가된 MFCC (환경 변화 체감)
    plt.subplot(3, 1, 2)
    librosa.display.specshow(mfcc_noisy, sr=sr, x_axis='time')
    plt.title('Noisy MFCC (Environment Change)')
    plt.colorbar()

    # 짧은 구간 MFCC (발화 길이 문제 체감)
    plt.subplot(3, 1, 3)
    librosa.display.specshow(mfcc_short, sr=sr, x_axis='time')
    plt.title('Short-utterance MFCC (Data Scarcity)')
    plt.colorbar()

    plt.tight_layout()
    plt.show()

    # 유사도 수치 계산 (원본과 얼마나 달라졌는가?)
    dist_noisy = np.linalg.norm(np.mean(mfcc_orig, axis=1) - np.mean(mfcc_noisy, axis=1))
    print(f"소음 추가 시 특징 변화 거리: {dist_noisy:.2f}")

# 실행
simulate_technical_challenges(y, sr)

```
