## 2. 실습 코드 (Python)

가장 대표적인 **WER(단어 오류율)** 계산과 **분류 평가지표**를 산출하는 파이썬 코드입니다.

### ① ASR 평가지표: WER 계산하기

한국어는 띄어쓰기 기준으로 단어를 분리하여 계산하는 것이 일반적입니다.

```python
def calculate_wer(reference, hypothesis):
    """
    WER = (S + D + I) / N
    S: 대체 (Substitution)
    D: 삭제 (Deletion)
    I: 삽입 (Insertion)
    """
    ref_words = reference.split()
    hyp_words = hypothesis.split()
    
    # 레벤슈타인 거리 알고리즘을 이용한 편집 거리 계산 (단어 단위)
    import numpy as np
    d = np.zeros((len(ref_words) + 1, len(hyp_words) + 1), dtype=np.uint8)
    
    for i in range(len(ref_words) + 1):
        d[i][0] = i
    for j in range(len(hyp_words) + 1):
        d[0][j] = j
        
    for i in range(1, len(ref_words) + 1):
        for j in range(1, len(hyp_words) + 1):
            if ref_words[i-1] == hyp_words[j-1]:
                d[i][j] = d[i-1][j-1]
            else:
                substitution = d[i-1][j-1] + 1
                insertion = d[i][j-1] + 1
                deletion = d[i-1][j] + 1
                d[i][j] = min(substitution, insertion, deletion)
    
    wer = d[len(ref_words)][len(hyp_words)] / len(ref_words)
    return wer

# 이미지 예시 적용
ref = "안녕하세요" # 참 값 (이미지 상 "안녕 하세요"로 분리 가정)
hyp = "안녕 하세요" # 인식 결과 (공백 포함 시 대체/삽입 발생)

# 이미지의 예시처럼 '안녕 하세요'를 2단어로 보고 '안녕하세요'를 1단어로 인식한 경우
print(f"WER: {calculate_wer('안녕 하세요', '안녕하세요') * 100}%") 

```

### ② 분류/감정 분석 평가지표

`scikit-learn` 라이브러리를 사용하여 정확도, F1 점수, 혼동 행렬을 간단히 구할 수 있습니다.

```python
from sklearn.metrics import accuracy_score, f1_score, confusion_matrix, classification_report

# 실제 정답(y_true)과 모델의 예측값(y_pred)
y_true = [0, 1, 0, 0, 1, 2, 2] # 0: 중립, 1: 긍정, 2: 부정
y_pred = [0, 2, 0, 0, 1, 2, 1]

# 1. 정확도 (Accuracy)
acc = accuracy_score(y_true, y_pred)

# 2. F1 점수 (F1 Score) - 다중 클래스일 경우 'weighted' 또는 'macro' 사용
f1 = f1_score(y_true, y_pred, average='weighted')

# 3. 혼동 행렬 (Confusion Matrix)
cm = confusion_matrix(y_true, y_pred)

print(f"정확도: {acc:.2f}")
print(f"F1 점수: {f1:.2f}")
print("\n[혼동 행렬]")
print(cm)

# 전체 리포트 출력
print("\n[상세 리포트]")
print(classification_report(y_true, y_pred))

```
