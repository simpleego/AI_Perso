### 2. 멀티모달 감정 분석 기초 실습 코드

이 코드는 음성(에너지), 텍스트(키워드), 영상(프레임 변화) 특징을 각각 추출한 뒤, 이를 결합하여 최종 감정을 추론하는 시뮬레이션 구조입니다.

```python
import librosa
import numpy as np
import cv2

# 실습을 위한 가상 데이터 설정
# 실제 환경에서는 각 파일을 로드하여 사용합니다.
audio_path = librosa.ex('choice') # 샘플 음성
video_path = None # 실제 영상 파일 경로 필요

def multimodal_fusion_analysis(audio_file, text_input):
    print("🚀 [Multimodal Pipeline] 분석을 시작합니다...\n")

    # --- 1. Audio Feature Extraction (음성) ---
    y, sr = librosa.load(audio_file, duration=3)
    rms = librosa.feature.rms(y=y)
    avg_energy = np.mean(rms)
    # 에너지가 높으면 활발함(High Arousal), 낮으면 침체됨(Low Arousal)
    audio_score = 1 if avg_energy > 0.03 else 0
    print(f"1. Audio 분석: 에너지 지수 {avg_energy:.4f} -> {'격양/활발' if audio_score else '차분/침체'}")

    # --- 2. Text Feature Extraction (텍스트) ---
    positive_dict = ['happy', 'good', 'great', 'love']
    negative_dict = ['bad', 'sad', 'angry', 'hate']
    
    text_score = 0
    words = text_input.lower().split()
    for w in words:
        if w in positive_dict: text_score = 1
        elif w in negative_dict: text_score = -1
    print(f"2. Text 분석: 문맥 감정 -> {'긍정' if text_score > 0 else '부정' if text_score < 0 else '중립'}")

    # --- 3. Video Feature Simulation (영상 - 밝기 변화율로 움직임 대체) ---
    # 실제 프로젝트에서는 얼굴 랜드마크(MediaPipe 등)를 사용합니다.
    video_score = 1 # 시뮬레이션: 표정 변화가 크다고 가정
    print(f"3. Video 분석: 표정 변화율 측정 완료 -> {'동적' if video_score else '정적'}")

    # --- 4. Feature Fusion & Classification (이미지의 'Classify' 단계) ---
    print("\n--- [Final Fusion Result] ---")
    
    # 간단한 논리 결합 (Fusion Logic)
    if text_score == 1 and audio_score == 1:
        final_emotion = "Joy (기쁨)"
    elif text_score == -1 and audio_score == 1:
        final_emotion = "Anger (분노)"
    elif text_score == -1 and audio_score == 0:
        final_emotion = "Sadness (슬픔)"
    else:
        final_emotion = "Neutral (평온)"

    print(f"최종 예측 감정: {final_emotion}")

# 실행 예시
user_text = "I am so angry about this situation"
multimodal_fusion_analysis(audio_path, user_text)

```
