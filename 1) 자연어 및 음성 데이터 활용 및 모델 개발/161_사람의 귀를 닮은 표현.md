## 1. 주파수 해상도 불균등
- 인간 귀는 **저주파에 민밀**하고 **고주파에 둔감**
- 고주파에서는 넓은 대역을 뭉뚱그려 인지 → 동일한 해상도가 비효율적

---

## 2. 멜(Mel) / 바크(Bark) 스케일
- 인간 청각의 **비선형 주파수 인지 모델**
- 멜 스케일 변환 공식:

$$
m(f) = 2595 \cdot \log_{10} \left(1 + \frac{f}{700}\right)
$$

- **MFCC**, **멜 스펙트로그램** → 음성 인식의 핵심 특징

---

## 3. 마스킹 현상
- 큰 소리가 가까운 주파수대의 작은 소리를 가려서 인지 불가
- **MP3, AAC** 등의 손실 압축 코덱 원리

---

## 4. 로그 스케일 필요성 (음량)
- 주관적 라우드니스 ≈ 물리적 음압의 로그
- 데시벨 정의:

$$
L_{\text{dB}} = 20 \log_{10} \left( \frac{P}{P_0} \right)
$$

- 인간 인지에 맞춘 표현 → 음량 비교가 직관적

---

## 결론
MFCC, 멜 스펙트로그램은 **단순 스펙트럼보다 음성/오디오 인식 성능이 우수**하며,  
이는 인간 청각의 **주파수 비선형성 + 마스킹 + 로그 음량 감각**을 반영했기 때문입니다.

---

## 1. 멜 스케일 변환 시각화

```python
import numpy as np
import matplotlib.pyplot as plt

def hz_to_mel(f):
    """Hz → Mel 변환"""
    return 2595 * np.log10(1 + f / 700)

def mel_to_hz(m):
    """Mel → Hz 변환"""
    return 700 * (10**(m / 2595) - 1)

# 주파수 범위
freqs = np.linspace(20, 8000, 1000)
mels = hz_to_mel(freqs)

# 시각화
plt.figure(figsize=(12, 4))

plt.subplot(1, 2, 1)
plt.plot(freqs, mels)
plt.xlabel('Frequency (Hz)')
plt.ylabel('Mel')
plt.title('Hz → Mel 변환 곡선')
plt.grid(True)

plt.subplot(1, 2, 2)
plt.plot(mels, freqs)
plt.xlabel('Mel')
plt.ylabel('Frequency (Hz)')
plt.title('Mel → Hz 변환 곡선')
plt.grid(True)

plt.tight_layout()
plt.show()

# 특정 주파수의 Mel 값 확인
test_freqs = [100, 500, 1000, 2000, 4000, 8000]
print("Hz → Mel 변환 예시:")
for f in test_freqs:
    print(f"  {f:4d} Hz → {hz_to_mel(f):.1f} Mel")
```

## 2. MFCC 추출 및 시각화

```python
import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt

# 샘플 오디오 파일 로드 (또는 생성)
duration = 3
sr = 16000
t = np.linspace(0, duration, int(sr * duration))
# 440Hz + 880Hz + 잡음
audio = 0.5 * np.sin(2 * np.pi * 440 * t) + 0.3 * np.sin(2 * np.pi * 880 * t)
audio += 0.1 * np.random.randn(len(t))

# MFCC 추출
n_mfcc = 13
mfccs = librosa.feature.mfcc(y=audio, sr=sr, n_mfcc=n_mfcc)

# 시각화
plt.figure(figsize=(12, 8))

plt.subplot(2, 1, 1)
librosa.display.waveshow(audio, sr=sr)
plt.title('Original Audio Waveform')
plt.xlabel('Time (s)')

plt.subplot(2, 1, 2)
librosa.display.specshow(mfccs, x_axis='time', sr=sr)
plt.colorbar(label='MFCC Coefficient')
plt.title(f'MFCC (n={n_mfcc})')
plt.xlabel('Time (s)')
plt.ylabel('MFCC Index')

plt.tight_layout()
plt.show()

print(f"MFCC shape: {mfccs.shape}")  # (n_mfcc, n_frames)
```

## 3. 멜 스펙트로그램 vs 일반 스펙트로그램 비교

```python
import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt
import soundfile as sf

# 오디오 생성 (선형 주파수 sweep)
duration = 2
sr = 16000
t = np.linspace(0, duration, int(sr * duration))
frequency_sweep = np.linspace(20, 4000, len(t))
audio = 0.5 * np.sin(2 * np.pi * frequency_sweep * t)

# 스펙트로그램 계산
D = librosa.stft(audio)
D_db = librosa.amplitude_to_db(np.abs(D), ref=np.max)

# 멜 스펙트로그램 계산
mel_spec = librosa.feature.melspectrogram(y=audio, sr=sr, n_mels=128)
mel_spec_db = librosa.power_to_db(mel_spec, ref=np.max)

# 시각화
plt.figure(figsize=(14, 6))

plt.subplot(1, 2, 1)
librosa.display.specshow(D_db, sr=sr, x_axis='time', y_axis='hz')
plt.colorbar(label='Amplitude (dB)')
plt.title('Linear-frequency Spectrogram')
plt.ylim(0, 5000)

plt.subplot(1, 2, 2)
librosa.display.specshow(mel_spec_db, sr=sr, x_axis='time', y_axis='mel')
plt.colorbar(label='Amplitude (dB)')
plt.title('Mel-frequency Spectrogram')

plt.tight_layout()
plt.show()
```

## 4. 마스킹 효과 시뮬레이션

```python
import numpy as np
import matplotlib.pyplot as plt
import IPython.display as ipd

sr = 16000
duration = 1
t = np.linspace(0, duration, int(sr * duration))

# 강한 저주파 신호 (500Hz) + 약한 고주파 신호 (700Hz)
strong_freq = 500  # Hz
weak_freq = 700    # Hz

strong_signal = 0.8 * np.sin(2 * np.pi * strong_freq * t)
weak_signal = 0.05 * np.sin(2 * np.pi * weak_freq * t)

mixed = strong_signal + weak_signal

# 스펙트럼 분석
fft_strong = np.fft.rfft(strong_signal)
fft_weak = np.fft.rfft(weak_signal)
fft_mixed = np.fft.rfft(mixed)

freqs = np.fft.rfftfreq(len(t), 1/sr)

# 시각화
plt.figure(figsize=(12, 8))

plt.subplot(2, 1, 1)
plt.plot(freqs, np.abs(fft_strong), label=f'{strong_freq}Hz (strong)', linewidth=2)
plt.plot(freqs, np.abs(fft_weak) * 10, label=f'{weak_freq}Hz (weak, x10)', alpha=0.7)  # 시각적 비교 위해 10배
plt.xlabel('Frequency (Hz)')
plt.ylabel('Magnitude')
plt.title('Individual Signals Spectrum')
plt.legend()
plt.xlim(300, 900)
plt.grid(True)

plt.subplot(2, 1, 2)
plt.plot(freqs, np.abs(fft_mixed), label='Mixed Signal', color='green', linewidth=2)
plt.xlabel('Frequency (Hz)')
plt.ylabel('Magnitude')
plt.title('Mixed Signal Spectrum (Weak signal masked by nearby strong signal)')
plt.xlim(300, 900)
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.show()

# 오디오 재생 (주피터 환경에서)
print("강한 신호 (500Hz) + 약한 신호 (700Hz)")
ipd.display(ipd.Audio(mixed, rate=sr))
```

## 5. dB 변환 시각화 (로그 스케일)

```python
import numpy as np
import matplotlib.pyplot as plt

def to_db(amplitude, ref=1.0):
    """진폭 → dB 변환"""
    return 20 * np.log10(amplitude / ref + 1e-10)  # 0 방지

# 진폭 범위
amplitudes = np.linspace(0.001, 1, 100)
dbs = to_db(amplitudes, ref=1.0)

# 시각화
plt.figure(figsize=(12, 4))

plt.subplot(1, 2, 1)
plt.plot(amplitudes, dbs)
plt.xlabel('Linear Amplitude')
plt.ylabel('dB')
plt.title('Linear → dB Conversion')
plt.grid(True)

plt.subplot(1, 2, 2)
# 귀에 똑같이 들리는 주관적 라우드니스
loudness = np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0])
loudness_db = to_db(loudness)
perceived = np.linspace(0, 1, len(loudness))

plt.plot(loudness_db, perceived, 'o-')
plt.xlabel('dB')
plt.ylabel('Perceived Loudness (relative)')
plt.title('Perceived Loudness vs dB')
plt.grid(True)

plt.tight_layout()
plt.show()

print("\n진폭 → dB 예시:")
for amp in [0.01, 0.1, 0.5, 1.0]:
    print(f"  Amplitude = {amp:4.2f} → {to_db(amp):6.2f} dB")
```
