
### 2. 실습 코드

이 코드는 `librosa` 라이브러리를 활용하여 실제 음성 파일에서 **F0, 에너지, 발화 속도**를 추출하는 과정을 담고 있습니다.

```python
import librosa
import librosa.display
import numpy as np
import matplotlib.pyplot as plt

# 1. 음성 파일 로드 (샘플 데이터 활용)
y, sr = librosa.load(librosa.ex('choice'))

def analyze_paralinguistics(y, sr):
    print(f"--- 음성 특징 분석 결과 ---")

    # [특징 1] F0 (기초 주파수) 추출
    # pyin 알고리즘을 사용하여 음높이(Pitch)를 추정합니다.
    f0, voiced_flag, voiced_probs = librosa.pyin(y, fmin=librosa.note_to_hz('C2'), fmax=librosa.note_to_hz('C7'))
    avg_f0 = np.nanmean(f0) # 결측치 제외 평균 계산
    print(f"1. 평균 기초 주파수(F0): {avg_f0:.2f} Hz")

    # [특징 2] 에너지 (RMSE 기반 단기 에너지)
    # 이미지의 에너지 수식과 대응되는 RMS 에너지를 계산합니다.
    rms = librosa.feature.rms(y=y)
    avg_energy = np.mean(rms)
    print(f"2. 평균 단기 에너지(Energy): {avg_energy:.4f}")

    # [특징 3] 발화 속도 시뮬레이션 (간단한 버전)
    # 무음 구간을 제외한 소리 구간의 길이를 통해 속도를 유추합니다.
    intervals = librosa.effects.split(y, top_db=20) # 20dB 이상의 소리 구간 추출
    total_speech_len = sum([start - end for start, end in intervals]) / sr
    speaking_rate = len(intervals) / (len(y) / sr) # 초당 발화 마디 수
    print(f"3. 추정 발화 속도(Rate): {speaking_rate:.2f} segments/sec")

    # 시각화
    plt.figure(figsize=(12, 8))

    # 웨이브폼과 F0 표시
    plt.subplot(2, 1, 1)
    librosa.display.waveshow(y, sr=sr, alpha=0.5)
    plt.plot(librosa.times_like(f0), f0, label='F0 (Pitch)', color='r', linewidth=2)
    plt.title('Waveform & Fundamental Frequency (F0)')
    plt.legend()

    # RMS 에너지 표시
    plt.subplot(2, 1, 2)
    plt.semilogy(rms.T, label='RMS Energy', color='g')
    plt.title('Short-term Energy (RMS)')
    plt.legend()
    plt.tight_layout()
    plt.show()

# 실습 실행
analyze_paralinguistics(y, sr)

```
