```python
from langchain.agents import create_agent
from langchain.chat_models import init_chat_model
from langchain.tools import tool, ToolRuntime
from langgraph.checkpoint.memory import MemorySaver
from langgraph.store.memory import InMemoryStore
from datetime import datetime

# 구매 이력 저장 도구
from langchain_core.tools import tool
from langchain_core.runnables import RunnableConfig

@tool
def save_purchase(product_name: str, category: str, config: RunnableConfig):
    """구매한 상품을 이력에 저장합니다."""
    # config에서 user_id 추출 (invoke 호출 시 전달한 config 기준)
    user_id = config.get("user_id", "default")
    
    # store는 전역 변수로 선언되어 있다고 가정
    namespace = ("users", user_id)
    key = "purchase-history"
    
    # 기존 이력 조회
    existing = store.get(namespace, key)
    # 기존 데이터가 있으면 .value 속성 사용
    history = existing.value if existing else []
    
    # 새 구매 추가
    history.append({
        "product": product_name,
        "category": category,
        "date": "2026-07-01" # 필요시 날짜 로직 추가
    })
    
    # Store에 저장 (튜플 namespace 사용)
    store.put(namespace, key, history)
    
    return f"{product_name} 상품을 구매 이력에 저장했습니다."

# 추천 상품 조회 도구
@tool
def get_recommendations(config: RunnableConfig):
    """사용자 이력 기반으로 상품을 추천합니다."""
    user_id = config.get("user_id", "default")
    namespace = ("users", user_id)
    key = "purchase-history"
    
    # store에서 가져온 결과가 Item 객체이므로 .value 속성 사용
    memory = store.get(namespace, key)
    
    if not memory:
        return "구매 이력이 없습니다."
    
    # 수정 부분: memory['value'] 대신 memory.value 사용
    history = memory.value 
    
    # 카테고리별 빈도 계산 로직...
    categories = {}
    for item in history:
        cat = item["category"]
        categories[cat] = categories.get(cat, 0) + 1
        
    top_category = max(categories, key=categories.get)
    return f"이전에 {top_category} 카테고리를 {categories[top_category]}번 구매하셨네요. {top_category} 관련 신상품을 추천드립니다."
    

# 에이전트 생성
model = init_chat_model("openai:gpt-4o")
tools = [save_purchase, get_recommendations]

store = InMemoryStore()
agent = create_agent(
    model,
    tools=tools,
    checkpointer=MemorySaver(),
    store=store
)

# 대화 1: 구매 기록
config = {
    "configurable": {"thread_id": "shopping-1"},
    "user_id": "user-789"
}

result = agent.invoke(
    {"messages": [("user", "노트북을 구매했어요, 카테고리는 전자제품입니다.")]},
    config=config
)
print(result["messages"][-1].content)

# 대화 2: 추가 구매
result = agent.invoke(
    {"messages": [("user", "마우스도 구매했습니다, 전자제품 카테고리요.")]},
    config=config
)
print(result["messages"][-1].content)

# 대화 3: 추천 요청
result = agent.invoke(
    {"messages": [("user", "뭔가 추천해줄 거 있나요?")]},
    config=config
)
print(result["messages"][-1].content)
# 출력: "이전에 전자제품 카테고리를 2번 구매하셨네요. 전자제품 관련 신상품을 추천드립니다."
```
