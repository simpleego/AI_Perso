```html
<!DOCTYPE html>
<html lang="ko">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>a태그 href 완성 및 이동</title>
    <style>
        body { font-family: sans-serif; margin: 30px; line-height: 1.6; }
        .form-group { margin-bottom: 15px; }
        label { display: inline-block; width: 80px; font-weight: bold; }
        input { padding: 8px; width: 250px; border: 1px solid #ccc; border-radius: 4px; }
        button { padding: 8px 15px; background-color: #28a745; color: white; border: none; border-radius: 4px; cursor: pointer; font-weight: bold; }
        button:hover { background-color: #218838; }
        .result-container { margin-top: 25px; padding: 20px; background-color: #f8f9fa; border: 1px solid #e9ecef; border-radius: 4px; }
        /* 이동 가능한 실제 링크 스타일 */
        .ready-link { display: inline-block; padding: 10px 20px; background-color: #007bff; color: white; text-decoration: none; border-radius: 4px; font-weight: bold; }
        .ready-link:hover { background-color: #0069d9; }
    </style>
</head>
<body>

    <h2>검색어 입력 후 링크 이동하기</h2>

    <!-- 1. 사용자 입력 양식 -->
    <div class="form-group">
        <label for="searchQuery">검색어:</label>
        <input type="text" id="searchQuery" placeholder="예: 치킨 맛집">
    </div>
    
    <div class="form-group">
        <label for="userLanguage">언어:</label>
        <input type="text" id="userLanguage" value="ko" placeholder="ko">
    </div>

    <!-- 2. 주소 완성 버튼 -->
    <button onclick="makeLinkAndReady()">1단계: 주소 완성하기</button>

    <!-- 3. 결과 출력 영역 (a태그 포함) -->
    <div class="result-container">
        <p><strong>2단계: 아래 버튼을 클릭하여 완성된 주소로 이동하세요.</strong></p>
        
        <!-- 최초 상태는 작동하지 않는 상태이며, 버튼을 누르면 JS가 href를 채워 넣습니다 -->
        <a id="targetLink" href="javascript:void(0);" onclick="alert('먼저 주소 완성하기 버튼을 눌러주세요.');" style="color: #6c757d; text-decoration: none;">
            [아직 주소가 완성되지 않았습니다]
        </a>
        
        <p style="margin-top: 15px; font-size: 0.9em; color: #666;">
            현재 a 태그의 href 실제 값: <code id="debugHref" style="background: #e9ecef; padding: 2px 6px;">없음</code>
        </p>
    </div>

    <script>
        function makeLinkAndReady() {
            // 사용자가 입력한 값 가져오기
            const query = document.getElementById('searchQuery').value.trim();
            const lang = document.getElementById('userLanguage').value.trim();

            if (!query) {
                alert('검색어를 입력해주세요!');
                return;
            }

            // URLSearchParams를 활용한 안전한 쿼리스트링 생성
            const params = new URLSearchParams();
            params.append('q', query);      // 구글 검색어 파라미터는 'q'
            params.append('hl', lang);     // 구글 언어 설정 파라미터는 'hl'

            // 실제 요청을 보낼 주소 (구글 검색 페이지)
            const baseUrl = "https://google.com";
            const completeUrl = baseUrl + "?" + params.toString();

            // a 태그 엘리먼트 가져오기
            const link = document.getElementById('targetLink');
            const debugText = document.getElementById('debugHref');

            // 1. a 태그의 href 속성에 완성된 주소 주입
            link.href = completeUrl;
            
            // 2. 새 창에서 열리도록 속성 추가 (선택 사항)
            link.target = "_blank"; 
            
            // 3. 임시로 막아둔 경고창(onclick) 속성 제거
            link.removeAttribute('onclick'); 
            
            // 4. 사용자가 알아보기 쉽게 디자인 및 텍스트 변경
            link.className = "ready-link";
            link.textContent = `'${query}' 결과 보러가기 (클릭)`;

            // 화면에 현재 완성된 href 주소 시각적으로 표시
            debugText.textContent = completeUrl;
        }
    </script>
</body>
</html>

```
