```python
import numpy as np
import matplotlib.pyplot as plt
import librosa
import librosa.display

# 1. 샘플 오디오 데이터 로드 (Librosa 샘플 음원 중 오페라/클래식 테마 사용)
# y: 오디오의 1차원 시계열 진폭 배열, sr: 샘플링 레이트 (기본값: 22050Hz)
audio_path = librosa.example('nutcracker')
y, sr = librosa.load(audio_path, duration=4.0)  # 직관적인 확인을 위해 앞부분 4초만 슬라이싱

# 2. 오디오 분석을 위한 핵심 하이퍼파라미터 정의
n_fft = 2048          # FFT 창의 크기 (주파수 해상도 결정)
hop_length = 512      # 창을 이동시키는 간격 (Overlap = n_fft - hop_length = 1536샘플)
n_mels = 128          # 멜 스펙트로그램의 Y축 필터 개수
n_mfcc = 20           # 추출할 고차원 MFCC 계수의 개수

# ==========================================
# [Feature 1] Short-Time Fourier Transform (스펙트로그램)
# ==========================================
stft_complex = librosa.stft(y, n_fft=n_fft, hop_length=hop_length, window='hann')
stft_amplitude = np.abs(stft_complex)
spectrogram_db = librosa.amplitude_to_db(stft_amplitude, ref=np.max)

# ==========================================
# [Feature 2] Mel-Spectrogram (멜 스펙트로그램)
# ==========================================
# 오디오 신호로부터 직접 구하거나, 이미 계산된 stft의 파워 스펙트럼(|STFT|^2)을 이용해 연산 가능
mel_spectrogram = librosa.feature.melspectrogram(y=y, sr=sr, n_fft=n_fft, 
                                                 hop_length=hop_length, n_mels=n_mels)
mel_db = librosa.power_to_db(mel_spectrogram, ref=np.max)

# ==========================================
# [Feature 3] MFCC (Mel-Frequency Cepstral Coefficients)
# ==========================================
# 멜 스펙트로그램 정보를 바탕으로 독립적인 Cepstral 계수를 DCT 연산으로 추출
mfcc = librosa.feature.mfcc(S=mel_db, n_mfcc=n_mfcc)

# ==========================================
# 4. 시각화 및 비교 분석 그래프 생성
# ==========================================
plt.figure(figsize=(14, 12))

# Subplot 1: 원시 파형 (Time Domain Waveform)
plt.subplot(4, 1, 1)
librosa.display.waveshow(y, sr=sr, color='royalblue')
plt.title('1. Original Waveform (Time Domain)', fontsize=12, fontweight='bold')
plt.ylabel('Amplitude')
plt.grid(True, linestyle='--', alpha=0.5)

# Subplot 2: 일반 스펙트로그램 (Linear Frequency Scale)
plt.subplot(4, 1, 2)
librosa.display.specshow(spectrogram_db, sr=sr, hop_length=hop_length, 
                         x_axis='time', y_axis='linear', cmap='magma')
plt.colorbar(format='%+2.0f dB')
plt.title('2. Linear Spectrogram (STFT Log-Magnitude)', fontsize=12, fontweight='bold')
plt.ylabel('Hz')

# Subplot 3: 멜 스펙트로그램 (Mel Frequency Scale)
plt.subplot(4, 1, 3)
librosa.display.specshow(mel_db, sr=sr, hop_length=hop_length, 
                         x_axis='time', y_axis='mel', cmap='viridis')
plt.colorbar(format='%+2.0f dB')
plt.title('3. Mel-Spectrogram (Human Hearing Scale)', fontsize=12, fontweight='bold')
plt.ylabel('Mel Hz')

# Subplot 4: MFCC (Cepstral Domain)
plt.subplot(4, 1, 4)
librosa.display.specshow(mfcc, sr=sr, hop_length=hop_length, x_axis='time', cmap='coolwarm')
plt.colorbar()
plt.title('4. MFCC (De-correlated Features via DCT)', fontsize=12, fontweight='bold')
plt.ylabel('MFCC Coeffs Index')

plt.tight_layout()
plt.show()

# ==========================================
# 5. 딥러닝 모델 관점의 데이터 Shape 교육용 출력
# ==========================================
print("\n" + "="*50)
print(" [Deep Learning Data Pipeline Dimension Check] ")
print("="*50)
print(f"▣ 1차원 원본 오디오 파형 Shape      : {y.shape}")
print(f"▣ 일반 스펙트로그램 (CNN Input 1)   : {spectrogram_db.shape} -> (주파수 빈 수, 시간 프레임 수)")
print(f"▣ 멜 스펙트로그램   (CNN Input 2)   : {mel_db.shape} -> (Mel 필터 수, 시간 프레임 수)")
print(f"▣ MFCC 특징 맵      (ML/RNN Input)  : {mfcc.shape} -> (추출 계수 차원, 시간 프레임 수)")
print("="*50)
print("Tip: 시간 프레임 수(가로축 크기)가 모두 일치하는 이유는 동일한 hop_length를 공유하기 때문입니다.")


```
