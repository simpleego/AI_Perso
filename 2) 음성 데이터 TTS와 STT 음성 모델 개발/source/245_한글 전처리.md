```python
import re
from konlpy.tag import Okt
from g2pk import G2p

class KoreanTextPreprocessor:
    def __init__(self):
        # 한국어 형태소 분석기 초기화
        self.okt = Okt()
        # 한국어 G2P 변환기 초기화
        self.g2p = G2p()
        
    def text_normalization(self, text: str) -> str:
        """
        1단계: 간단한 텍스트 정규화 (Text Normalization)
        단순 특수문자 제거 및 문맥 외 기호 정리
        """
        # 영어, 한글, 숫자, 주요 구두점(.,!?)을 제외한 불필요한 특수문자 제거
        normalized_text = re.sub(r'[^a-zA-Z0-9가-힣\s.,!?~]', '', text)
        return normalized_text

    def analyze_morphemes(self, text: str):
        """
        2단계: 형태소 분석 (Morpheme Analysis)
        어떤 단어가 실질 형태소인지, 형식 형태소(조사 등)인지 분리하여 확인
        """
        return self.okt.pos(text)

    def convert_g2p(self, text: str) -> str:
        """
        3단계: G2P 변환 (Grapheme-to-Phoneme)
        텍스트 정규화(숫자 한글화 포함)와 의문문 등의 억양을 고려한 발음 기호 변환
        """
        # g2pk는 내부에 숫자 확장(2026 -> 이천이십육) 기능과 
        # 한국어 표준 발음법 규칙(연음, 동화 등)을 모두 포함하고 있습니다.
        return self.g2p(text)

    def run_pipeline(self, raw_text: str):
        """
        전체 전처리 파이프라인 실행 및 시각화
        """
        print(f"==================================================")
        print(f"[입력 원문]: {raw_text}\n")
        
        # 1. 정규화
        norm_text = self.text_normalization(raw_text)
        print(f"[1단계] 텍스트 정규화 결과:")
        print(f"   -> {norm_text}\n")
        
        # 2. 형태소 분석
        morphemes = self.analyze_morphemes(norm_text)
        print(f"[2단계] 형태소 분석 (품사 태깅):")
        print(f"   -> {morphemes[:8]} ... (하단 상세 확인)\n")
        
        # 3. G2P 변환 (최종 TTS 입력 형태)
        phoneme_text = self.convert_g2p(norm_text)
        print(f"[3단계] 최종 G2P 발음 변환 결과 (TTS 모델 입력값):")
        print(f"   -> {phoneme_text}")
        print(f"==================================================\n")


# --- 실제 테스트 실행 ---
if __name__ == "__main__":
    preprocessor = KoreanTextPreprocessor()
    
    # 테스트 케이스 1: 숫자가 포함되고 연음/조사 구개음화가 일어나는 문장
    # '밭이' -> [바치] (구개음화), '국내' -> [궁내] (자음동화)
    text_1 = "2026년 5월 19일, 밭이 기름진 국내 농가를 방문했다@#$!"
    preprocessor.run_pipeline(text_1)
    
    # 테스트 케이스 2: 동일한 글자('맛')가 문맥(실질/형식 형태소)에 따라 다르게 발음되는 경우
    # '맛있다' -> [마싣따], '맛없다' -> [마덥따]
    text_2 = "이 음식은 맛있다. 하지만 저 음식은 맛없다."
    preprocessor.run_pipeline(text_2)

```
