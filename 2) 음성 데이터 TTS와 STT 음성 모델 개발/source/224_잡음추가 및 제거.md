```python
import numpy as np
import matplotlib.pyplot as plt
import librosa
import librosa.display
import scipy.signal as signal

# ==========================================
# 0. 깨끗한 원본 음성 데이터 로드
# ==========================================
audio_path = librosa.example('libri1')  # 깨끗한 독백 음성 샘플
y_clean, sr = librosa.load(audio_path, duration=3.0)

# ==========================================
# STEP 1. 잡음 추가 (White Gaussian Noise 주입)
# ==========================================
np.random.seed(42)  # 결과 재현을 위한 시드 고정
noise = np.random.normal(0, 1, len(y_clean))

# ==========================================
# STEP 2. 목표 SNR 기준 신호 합성 및 SNR 계산 함수
# ==========================================
def padding_noise_to_target_snr(clean_signal, noise_signal, target_snr_db=5):
    # 각 신호의 전력(Power) 계산
    power_clean = np.mean(clean_signal ** 2)
    power_noise = np.mean(noise_signal ** 2)
    
    # 목표 SNR을 만족하기 위한 잡음의 스케일링 가중치 계산
    # SNR = 10 * log10(P_clean / (k^2 * P_noise)) -> k 계산
    k = np.sqrt(power_clean / (10 ** (target_snr_db / 10) * power_noise))
    
    scaled_noise = k * noise_signal
    noisy_signal = clean_signal + scaled_noise
    
    # 실제 합성된 신호의 SNR 역산 검증
    actual_snr = 10 * np.log10(np.mean(clean_signal ** 2) / np.mean(scaled_noise ** 2))
    return noisy_signal, scaled_noise, actual_snr

target_snr = 5  # 5dB 수준의 강한 노이즈 주입 설정
y_noisy, scaled_noise, calculated_snr = padding_noise_to_target_snr(y_clean, noise, target_snr)

# ==========================================
# STEP 3. 스펙트럼 차감법 (Spectral Subtraction) 구현
# ==========================================
def spectral_subtraction(noisy_signal, n_fft=1024, hop_length=256):
    # 1. 주파수 도메인 변환
    stft_noisy = librosa.stft(noisy_signal, n_fft=n_fft, hop_length=hop_length)
    magnitude_noisy = np.abs(stft_noisy)
    phase_noisy = np.angle(stft_noisy)
    
    # 2. 잡음 프로파일 추정 (신호의 초기 10개 프레임을 순수 잡음 구간으로 가정)
    noise_profile = np.mean(magnitude_noisy[:, :10], axis=1, keepdims=True)
    
    # 3. 차감 수행 (과도한 음수 값 발생 방지를 위한 제어 계수 적용)
    alpha = 2.0  # 과도 차감 계수 (Over-subtraction factor)
    beta = 0.02  # 플로어링 계수 (Spectral floor)
    
    magnitude_estimated = magnitude_noisy - alpha * noise_profile
    magnitude_estimated = np.maximum(magnitude_estimated, beta * noise_profile)
    
    # 4. 원래의 위상 정보와 결합 후 시간 도메인 복원
    stft_denoised = magnitude_estimated * np.exp(1j * phase_noisy)
    return librosa.istft(stft_denoised, hop_length=hop_length)

y_denoised_spectral = spectral_subtraction(y_noisy)

# ==========================================
# STEP 4. 위너 필터링 (Wiener Filtering) 구현
# ==========================================
# Scipy의 주파수 영역 위너 필터 최적화 함수 활용 (scipy.signal.wiener)
# 시간 도메인에서 로컬 분산을 기반으로 노이즈를 평활화하는 필터
y_denoised_wiener = signal.wiener(y_noisy, mysize=15)

# ==========================================
# STEP 5. 결과 평가 (SNR 개선도 계산)
# ==========================================
def compute_snr(clean, noisy_or_denoised):
    error_noise = noisy_or_denoised - clean
    # 분모가 0이 되는 것을 방지하기 위해 미소값 추가
    snr = 10 * np.log10(np.mean(clean ** 2) / (np.mean(error_noise ** 2) + 1e-10))
    return snr

snr_initial = compute_snr(y_clean, y_noisy)
snr_after_spectral = compute_snr(y_clean, y_denoised_spectral)
snr_after_wiener = compute_snr(y_clean, y_denoised_wiener)

# ==========================================
# 6. 시각화 및 비교 검증
# ==========================================
plt.figure(figsize=(15, 10))

# 1열: 파형 플롯 비교
plt.subplot(3, 2, 1)
librosa.display.waveshow(y_clean, sr=sr, color='g')
plt.title('Clean Original Waveform')
plt.ylabel('Amplitude')

plt.subplot(3, 2, 3)
librosa.display.waveshow(y_noisy, sr=sr, color='r')
plt.title(f'Noisy Waveform (Synthesized SNR: {snr_initial:.2f} dB)')
plt.ylabel('Amplitude')

plt.subplot(3, 2, 5)
librosa.display.waveshow(y_denoised_spectral, sr=sr, color='b')
plt.title(f'Denoised via Spectral Subtraction (SNR: {snr_after_spectral:.2f} dB)')
plt.ylabel('Amplitude')

# 2열: 스펙트로그램 관찰을 통한 주파수 정제 성능 확인
stft_clean = librosa.amplitude_to_db(np.abs(librosa.stft(y_clean)), ref=np.max)
stft_noisy = librosa.amplitude_to_db(np.abs(librosa.stft(y_noisy)), ref=np.max)
stft_spectral = librosa.amplitude_to_db(np.abs(librosa.stft(y_denoised_spectral)), ref=np.max)

plt.subplot(3, 2, 2)
librosa.display.specshow(stft_clean, sr=sr, x_axis='time', y_axis='linear', cmap='magma')
plt.title('Clean Spectrogram')

plt.subplot(3, 2, 4)
librosa.display.specshow(stft_noisy, sr=sr, x_axis='time', y_axis='linear', cmap='magma')
plt.title('Noisy Spectrogram')

plt.subplot(3, 2, 6)
librosa.display.specshow(stft_spectral, sr=sr, x_axis='time', y_axis='linear', cmap='magma')
plt.title('Spectral Subtracted Spectrogram')

plt.tight_layout()
plt.show()

# 최종 벤치마크 지표 출력
print("\n" + "="*50)
print(" [오디오 노이즈 제거 알고리즘 성능 평가 지표] ")
print("="*50)
print(f" 기준 목표 설정 SNR       : {target_snr}.00 dB")
print(f" 1. 초기 오염된 신호 SNR  : {snr_initial:.2f} dB")
print(f" 2. 스펙트럼 차감 후 SNR  : {snr_after_spectral:.2f} dB (개선도: {snr_after_spectral - snr_initial:+.2f} dB)")
print(f" 3. 위너 필터링 후 SNR    : {snr_after_wiener:.2f} dB (개선도: {snr_after_wiener - snr_initial:+.2f} dB)")
print("="*50)
```
