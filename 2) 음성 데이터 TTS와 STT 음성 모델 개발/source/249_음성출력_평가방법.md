```python
import numpy as np
import scipy.stats as stats

def analyze_mos_results(model_name, scores):
    """
    수집된 점수 배열을 바탕으로 평균(MOS)과 95% 신뢰구간을 계산합니다.
    """
    scores = np.array(scores)
    n = len(scores)
    mean = np.mean(scores)
    
    # 표준오차(Sample Standard Error of the Mean) 계산
    sem = stats.sem(scores)
    
    # 95% 신뢰구간의 임계값 계산 (t-분포 활용)
    confidence_interval = sem * stats.t.ppf((1 + 0.95) / 2., n - 1)
    
    print(f"[{model_name} MOS 분석 결과]")
    print(f"  - 평가 데이터 수 (U x N): {n}개")
    print(f"  - 최종 MOS 점수: {mean:.3f} ± {confidence_interval:.3f}")
    print(f"  - 95% 신뢰구간 범위: [{mean - confidence_interval:.3f} ~ {mean + confidence_interval:.3f}]")
    return mean, confidence_interval, scores

def run_cmos_and_ttest(scores_a, scores_b):
    """
    두 모델 간의 상대적 비교(CMOS)를 수행하고
    두 그룹의 차이가 통계적으로 정말 의미가 있는지 독립표본 t-검정(Two-sample t-test)을 진행합니다.
    """
    # 가상의 CMOS 계산 (B 모델 점수 - A 모델 점수)
    # 실제 CMOS는 평가자가 직접 -3 ~ +3 점수를 매기지만, 여기서는 통계적 유의성 검정을 병행합니다.
    t_stat, p_value = stats.ttest_ind(scores_b, scores_a, equal_var=False)
    
    print(f"\n[모델 간 통계적 유의성 검정 (T-test)]")
    print(f"  - t-statistic: {t_stat:.3f}")
    print(f"  - p-value: {p_value:.5f}")
    
    # 일반적인 유의수준 0.05 기준 판정
    if p_value < 0.05:
        print("  -> 판정: p-value가 0.05보다 작으므로, 두 모델의 품질 차이는 통계적으로 유의미합니다. (우연이 아님)")
    else:
        print("  -> 판정: p-value가 0.05보다 크므로, 두 모델의 품질 차이는 통계적으로 우연일 확률이 높습니다. (성능 차이 미미)")

# --- 실제 평가 데이터 분석 시뮬레이션 ---
if __name__ == "__main__":
    print("==================================================")
    print("      TTS 합성음 품질 주관적 평가 통계 분석 실습      ")
    print("==================================================\n")
    
    # 1. 가상의 평가자 점수 데이터 데이터셋 준비 (1점~5점 척도)
    # 기존 모델 (예: Tacotron2 기반 기본 파이프라인)
    baseline_scores = [3, 4, 4, 3, 4, 5, 3, 4, 4, 3, 4, 3, 4, 4, 5, 3, 4, 3, 4, 4]
    
    # 개선 모델 (예: FastSpeech2 + HiFi-GAN 적용 파이프라인)
    proposed_scores = [4, 5, 5, 4, 4, 5, 4, 5, 4, 4, 5, 4, 5, 5, 4, 4, 5, 4, 5, 5]
    
    # 2. 각 모델별 MOS 및 95% 신뢰구간 산출 (수식 구현)
    mean_a, ci_a, data_a = analyze_mos_results("Baseline Model (A)", baseline_scores)
    print("-" * 50)
    mean_b, ci_b, data_b = analyze_mos_results("Proposed Model (B)", proposed_scores)
    
    # 3. 이미지에 등장한 수식 기반 대용 가상 CMOS 분석 및 유의성 검정
    # 실제 CMOS 척도 스타일 체킹 (B - A의 평균적인 편차 경향)
    pseudo_cmos = mean_b - mean_a
    print("\n" + "=" * 50)
    print(f"[상대 평가 지표 시뮬레이션]")
    print(f"  - 베이스라인(A) 대비 제안 모델(B)의 점수 우위량: +{pseudo_cmos:.3f}")
    
    run_cmos_and_ttest(data_a, data_b)
    print("==================================================")
```
