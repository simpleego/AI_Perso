import torch
import librosa
import time
from transformers import Wav2Vec2Processor, Wav2Vec2ForCTC
from jiwer import wer, cer
from nltk.translate.bleu_score import sentence_bleu, SmoothingFunction

# 1. 사전학습된 한국어 Wav2Vec2 모델 로드
model_name = "kresnik/wav2vec2-large-xlsr-korean"
processor = Wav2Vec2Processor.from_pretrained(model_name)
model = Wav2Vec2ForCTC.from_pretrained(model_name)

# 2. 오디오 파일 로드 및 정답 텍스트 설정
file_path = "sample01.wav"
reference_text = '그는 재능있는 작곡가로서 명성을 쌓아 나가고 있다' # 제공해주신 정답 텍스트

audio, rate = librosa.load(file_path, sr=16000)
audio_duration = librosa.get_duration(y=audio, sr=rate) # 오디오 길이 (초)

# 3. 입력 전처리 및 추론 (RTF 계산을 위해 시간 측정 시작)
start_time = time.time()

inputs = processor(audio, sampling_rate=16000, return_tensors="pt", padding=True)

with torch.no_grad():
    logits = model(inputs.input_values).logits

predicted_ids = torch.argmax(logits, dim=-1)

# 4. 토큰을 텍스트로 변환 (인식 결과 변수 할당)
transcription = processor.decode(predicted_ids[0])

# 추론 종료 시간 측정
end_time = time.time()
inference_time = end_time - start_time

print(f"정답: {reference_text}")
print(f"인식 결과: {transcription}")
print("-" * 30)

# 5. 평가지표 계산 로직 추가
# 1) WER (Word Error Rate) - 단어 에러율
error_wer = wer(reference_text, transcription)

# 2) CER (Character Error Rate) - 음절(문자) 에러율
error_cer = cer(reference_text, transcription)

# 3) RTF (Real Time Factor) - 실시간 처리 지수 (추론시간 / 오디오길이)
rtf = inference_time / audio_duration

# 4) BLEU Score - 기계 번역 평가 지표 (자연어 유사도)
# 한국어는 형태소 단위가 정확하나, 여기서는 간단히 어절 단위로 계산
ref_tokens = [reference_text.split()]
pred_tokens = transcription.split()
bleu_score = sentence_bleu(ref_tokens, pred_tokens, smoothing_function=SmoothingFunction().method1)

# 결과 출력
print(f"1. WER (Word Error Rate): {error_wer:.4f}")
print(f"2. CER (Character Error Rate): {error_cer:.4f}")
print(f"3. RTF (Real Time Factor): {rtf:.4f} (Inference: {inference_time:.2f}s / Audio: {audio_duration:.2f}s)")
print(f"4. BLEU Score: {bleu_score:.4f}")
