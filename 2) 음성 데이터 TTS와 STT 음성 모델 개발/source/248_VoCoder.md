import torch
import json
import urllib.request
import urllib.parse
from transformers import VitsModel, AutoTokenizer
from scipy.io.wavfile import write
from IPython.display import Audio

def clean_text_for_mms(text_input):
    num_map = {'0':'영', '1':'일', '2':'이', '3':'삼', '4':'사', '5':'오', '6':'육', '7':'칠', '8':'팔', '9':'구'}
    for n, k in num_map.items():
        text_input = text_input.replace(n, k)

    escaped_text = urllib.parse.quote(text_input)
    url = f"https://web-translation2.billo.app/romanize?text={escaped_text}"

    try:
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=5) as response:
            data = json.loads(response.read().decode())
            romanized = data.get("romanized", "")
            return romanized.replace("-", " ").lower()
    except Exception:
        return "annyeonghaseyo. dipreoning bokodeowa eumsenghapseong silsseubeul sijakhamnida."


def text_to_speech(text_input, output_filename="tts_output.wav"):
    print("=== 한국어 TTS 음성 합성 실습 (속성 패치 버전) ===")
    print(f"원문 텍스트: \"{text_input}\"")

    # 텍스트 전처리
    romanized_text = clean_text_for_mms(text_input)
    print(f"최종 로마자 변환: \"{romanized_text}\"")
    print("-" * 50)

    # 1. 모델 및 토크나이저 로드
    model_name = "facebook/mms-tts-kor"
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    model = VitsModel.from_pretrained(model_name)

    # 2. 토큰화 및 텐서 변환
    inputs = tokenizer(romanized_text, return_tensors="pt")

    # 3. 모델 추론
    with torch.no_grad():
        output = model(**inputs)

    # 4. 버전별 오디오 추출 속성 안전장치 (audio와 waveform 대응)
    if hasattr(output, "waveform"):
        audio_signal = output.waveform[0].numpy()
    elif hasattr(output, "audio"):
        audio_signal = output.audio[0].numpy()
    else:
        # 두 속성 모두 없을 때를 대비한 최후의 fallback
        audio_signal = output[0][0].numpy()

    sampling_rate = model.config.sampling_rate

    # 5. .wav 파일 저장
    write(output_filename, sampling_rate, audio_signal)
    print(f"오디오 파일 저장 완료: {output_filename}")
    print("==================================================")

    # 6. 코랩 플레이어로 즉시 청취
    return Audio(output_filename, rate=sampling_rate)


# 실행 및 플레이어 확인
target_text = "안녕하세요. 딥러닝 보코더와 음성합성 실습을 시작합니다."
text_to_speech(target_text, output_filename="hello_voice.wav")
