from transformers import AutoTokenizer, AutoModelForSeq2SeqLM
import torch

def translate_texts_mbart(texts, model_id, src_lang, tgt_lang):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

    tokenizer = AutoTokenizer.from_pretrained(model_id)
    model = AutoModelForSeq2SeqLM.from_pretrained(model_id).to(device)

    # Set source language for tokenizer
    tokenizer.src_lang = src_lang

    inputs = tokenizer(
        texts,
        return_tensors="pt",
        padding=True,
        truncation=True,
        max_length=512
    ).to(device)

    # Set forced_bos_token_id for target language during generation
    generated_tokens = model.generate(
        **inputs,
        forced_bos_token_id=tokenizer.lang_code_to_id[tgt_lang],
        max_length=128,
        num_beams=4,
        early_stopping=True
    )

    translated = tokenizer.batch_decode(generated_tokens, skip_special_tokens=True)
    return translated

# ==========================================
# MBart 실행 메인 로직
# ==========================================
print(f"현재 사용 중인 하드웨어: {torch.device('cuda') if torch.cuda.is_available() else 'cpu'}\n")

mbart_model_id = "facebook/mbart-large-50-many-to-many-mmt"

# [1] 영어 (en_XX) -> 한국어 (ko_KR) 번역
print("=== [MBart] 영어 -> 한국어 번역 ===")
en_texts_mbart = [
    "I watched a movie with my friend yesterday.",
    "Deep learning models are changing the job market."
]
ko_results_mbart = translate_texts_mbart(en_texts_mbart, mbart_model_id, "en_XX", "ko_KR")
for en, ko in zip(en_texts_mbart, ko_results_mbart):
    print(f"EN: {en}")
    print(f"KO: {ko}\n")

print("-" * 50)

# [2] 한국어 (ko_KR) -> 영어 (en_XX) 번역
print("=== [MBart] 한국어 -> 영어 번역 ===")
ko_texts_mbart = [
    # "어제 친구와 함께 재미있는 영화를 봤습니다.",
    "12·3 내란의 명분을 만들 목적으로 북한 평양에 무인기 침투 작전을 지시한 혐의를 받는 윤석열 전 대통령이 1심에서 징역 30년을 선고받았다. 서울중앙지법 형사합의36부(재판장 이정엽)는 12일 일반이적·직권남용권리행사방해 혐의를 받는 윤 전 대통령에게 징역 30년을 선고했다.",
    "Google Colab에서 언어 모델을 실행하고 있습니다."
]
en_results_mbart = translate_texts_mbart(ko_texts_mbart, mbart_model_id, "ko_KR", "en_XX")
for ko, en in zip(ko_texts_mbart, en_results_mbart):
    print(f"KO: {ko}")
    print(f"EN: {en}\n")
