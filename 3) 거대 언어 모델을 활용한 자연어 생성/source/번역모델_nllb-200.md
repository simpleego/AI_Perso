from transformers import AutoTokenizer, AutoModelForSeq2SeqLM
import torch

def translate_texts_mbart(texts, model_id, src_lang, tgt_lang):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

    tokenizer = AutoTokenizer.from_pretrained(model_id)
    model = AutoModelForSeq2SeqLM.from_pretrained(model_id).to(device)

    # Set source language for tokenizer
    tokenizer.src_lang = src_lang

    inputs = tokenizer(
        texts,
        return_tensors="pt",
        padding=True,
        truncation=True,
        max_length=512
    ).to(device)

    # Set forced_bos_token_id for target language during generation
    generated_tokens = model.generate(
        **inputs,
        forced_bos_token_id=tokenizer.lang_code_to_id[tgt_lang],
        max_length=128,
        num_beams=4,
        early_stopping=True
    )

    translated = tokenizer.batch_decode(generated_tokens, skip_special_tokens=True)
    return translated

# [1] 영어 (en_XX) -> 한국어 (ko_KR) 번역

en_texts_mbart = [
    "I watched a movie with my friend yesterday.",

ko_results_mbart = translate_texts_mbart(en_texts_mbart, mbart_model_id, "en_XX", "ko_KR")

"""### Translation using NLLB Model (e.g., `facebook/nllb-200-distilled-600M`)

Another excellent multilingual translation model is `facebook/nllb-200-distilled-600M` from the NLLB project. It also supports 200 languages and requires specifying the source and target language codes. The language codes for NLLB are slightly different (e.g., `eng_Latn` for English, `kor_Kore` for Korean).
"""

from transformers import AutoTokenizer, AutoModelForSeq2SeqLM
import torch

def translate_texts_nllb(texts, model_id, src_lang_code, tgt_lang_code):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

    tokenizer = AutoTokenizer.from_pretrained(model_id)
    model = AutoModelForSeq2SeqLM.from_pretrained(model_id).to(device)

    # Set source language for tokenizer (using NLLB-specific codes)
    tokenizer.src_lang = src_lang_code

    inputs = tokenizer(
        texts,
        return_tensors="pt",
        padding=True,
        truncation=True,
        max_length=512
    ).to(device)

    # For NLLB, use convert_tokens_to_ids to get the forced_bos_token_id
    forced_bos_token_id = tokenizer.convert_tokens_to_ids(tgt_lang_code)

    generated_tokens = model.generate(
        **inputs,
        forced_bos_token_id=forced_bos_token_id,
        max_length=128,
        num_beams=4,
        early_stopping=True,
    )

    translated = tokenizer.batch_decode(generated_tokens, skip_special_tokens=True)
    return translated

# [1] 영어 (eng_Latn) -> 한국어 (kor_Kore) 번역
