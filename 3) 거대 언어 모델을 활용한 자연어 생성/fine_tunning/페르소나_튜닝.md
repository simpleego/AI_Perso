```python
import torch
from datasets import Dataset
from transformers import AutoTokenizer, AutoModelForCausalLM, BitsAndBytesConfig
from peft import LoraConfig, prepare_model_for_kbit_training, get_peft_model
from trl import SFTTrainer, SFTConfig

print("🚀 페르소나 AI 파인튜닝을 시작합니다...")

# ==========================================
# [1단계] 페르소나 학습 데이터셋 준비
# ==========================================
# 예시 페르소나: "까칠하지만 속은 따뜻한 서울 아저씨"
# 실제 실무에서는 이 데이터를 50~100개 이상 준비해야 합니다.
persona_data = [
    {
        "system": "당신은 말은 좀 까칠하지만 속은 따뜻한 서울 아저씨입니다. 반말을 섞어서 답변하세요.",
        "user": "안녕하세요! 오늘 기분이 어때요?",
        "assistant": "야, 안녕하세요는 무슨. 그냥 그래. 너는 뭐 하냐? 밥은 먹었어? 안 먹었으면赶紧(赶紧) 가서 뭐든 좀 처먹어."
    },
    {
        "system": "당신은 말은 좀 까칠하지만 속은 따뜻한 서울 아저씨입니다. 반말을 섞어서 답변하세요.",
        "user": "아저씨, 제가 이번 시험에 떨어졌어요. 너무 슬퍼요.",
        "assistant": "아이고, 떨어져서 슬프면 어떡해. 떨어지면 다시 붙으면 되지. 울고 있을 시간에 책이나 펴. 내가 맛있는 거 사줄게, 일어나."
    },
    {
        "system": "당신은 말은 좀 까칠하지만 속은 따뜻한 서울 아저씨입니다. 반말을 섞어서 답변하세요.",
        "user": "오늘 저녁 메뉴 뭐가 좋을까요?",
        "assistant": "그건 니가 알아서 정해라. 나는 뭐든 잘 먹어. 근데 날씨가 좀 쌀쌀하니까 뜨끈한 국물 있는 거나 처먹어라. 감기 걸린다."
    },
    {
        "system": "당신은 말은 좀 까칠하지만 속은 따뜻한 서울 아저씨입니다. 반말을 섞어서 답변하세요.",
        "user": "아저씨 사랑해요!",
        "assistant": "뭐, 뭐라는 거야. 갑자기. ...어, 나도. 얼른 가서 자라, 내일 일찍 일어나야 할 거 아니야."
    }
]

# Hugging Face Dataset으로 변환
dataset = Dataset.from_list(persona_data)

# ==========================================
# [2단계] 모델 및 토나이저 로드 (4bit 양자화)
# ==========================================
model_name = "Qwen/Qwen2.5-1.5B-Instruct"

# 4bit 양자화 설정 (메모리 절약의 마법)
bnb_config = BitsAndBytesConfig(
    load_in_4bit=True,
    bnb_4bit_quant_type="nf4",
    bnb_4bit_compute_dtype=torch.float16
)

tokenizer = AutoTokenizer.from_pretrained(model_name)
# Qwen 모델은 pad_token이 eos_token과 같도록 설정해야 에러가 나지 않습니다.
tokenizer.pad_token = tokenizer.eos_token 

model = AutoModelForCausalLM.from_pretrained(
    model_name,
    quantization_config=bnb_config,
    device_map="auto",
    torch_dtype=torch.float16
)

# ==========================================
# [3단계] LoRA 설정 (작은 어댑터 장착)
# ==========================================
model = prepare_model_for_kbit_training(model)

lora_config = LoraConfig(
    r=16,               # LoRA 크 (클수록 똑똑해지지만 무거워짐)
    lora_alpha=32,      # 학습 스케일
    target_modules=["q_proj", "k_proj", "v_proj", "o_proj", "gate_proj", "up_proj", "down_proj"],
    lora_dropout=0.05,
    bias="none",
    task_type="CAUSAL_LM"
)
model = get_peft_model(model, lora_config)
model.print_trainable_parameters() # 학습 가능한 파라미터 확인

# ==========================================
# [4단계] SFTTrainer 설정 및 학습 실행
# ==========================================
# Qwen의 ChatML 포맷을 적용하는 함수
def formatting_func(example):
    messages = [
        {"role": "system", "content": example["system"]},
        {"role": "user", "content": example["user"]},
        {"role": "assistant", "content": example["assistant"]}
    ]
    text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=False)
    return text

# 학습 설정
training_args = SFTConfig(
    output_dir="./qwen_persona_model",
    num_train_epochs=5,          # 데이터가 적으므로 Epoch를 조금 높게
    per_device_train_batch_size=2,
    gradient_accumulation_steps=4,
    learning_rate=2e-4,
    fp16=True,
    logging_steps=1,
    max_seq_length=512,
    report_to="none"
)

trainer = SFTTrainer(
    model=model,
    args=training_args,
    train_dataset=dataset,
    processing_class=tokenizer,
    formatting_func=formatting_func,
)

print("\n🔥 페르소나 학습(파인튜닝) 시작! (약 5~10분 소요)")
trainer.train()

# ==========================================
# [5단계] 학습된 모델 테스트
# ==========================================
print("\n" + "="*40)
print("🤖 페르소나 AI 테스트")
print("="*40)

model.eval()
test_prompt = "아저씨, 나 오늘 회사에서 혼났어. 너무 억울해."

# ChatML 포으로 프롬프트 생성
messages = [
    {"role": "system", "content": "당신은 말은 좀 까칠하지만 속은 따뜻한 서울 아저씨입니다. 반말을 섞어서 답변하세요."},
    {"role": "user", "content": test_prompt}
]
text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
inputs = tokenizer(text, return_tensors="pt").to(model.device)

with torch.no_grad():
    outputs = model.generate(**inputs, max_new_tokens=150, do_sample=True, temperature=0.7)

response = tokenizer.decode(outputs[0], skip_special_tokens=True)
# 프롬프트 부분을 제외하고 AI의 답변만 추출
final_answer = response.split("<|im_start|>assistant\n")[-1].strip()

print(f"👤 사용자: {test_prompt}")
print(f" 아저씨: {final_answer}")
```
