```python

import torch
from transformers import AutoTokenizer, AutoModelForCausalLM, pipeline
import gc

def clear_memory():
    """GPU 메모리를 비워 다음 모델이 원활하게 로드되도록 합니다."""
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()

# =====================================================================
# 1. Alpaca 모델 공략: 비즈니스 이메일 자동 생성 (Text Generation)
# =====================================================================
def run_alpaca_task():
    # 스탠퍼드 Alpaca 양식으로 파인튜닝된 대표적인 오픈소스 체크포인트
    model_id = "chavinlo/alpaca-native"
    print(f"\n[1/3] Alpaca 모델 로딩 중... (응용 분야: 이메일 생성)")

    tokenizer = AutoTokenizer.from_pretrained(model_id)
    model = AutoModelForCausalLM.from_pretrained(
        model_id, torch_dtype=torch.float16, device_map="auto"
    )

    # Alpaca 고유의 Instruction/Response 프롬프트 템플릿 구성
    alpaca_prompt = (
        "Below is an instruction that describes a task. "
        "Write a response that appropriately completes the request.\n\n"
        "### Instruction:\n"
        "Write a professional email to a client explaining that our server deployment "
        "will be delayed by 2 days due to unexpected database migration issues. "
        "Keep the tone polite and apologetic.\n\n"
        "### Response:\n"
    )

    generator = pipeline("text-generation", model=model, tokenizer=tokenizer)
    outputs = generator(alpaca_prompt, max_new_tokens=256, do_sample=True, temperature=0.7)

    print("\n✨ [Alpaca 결과] 비즈니스 이메일 생성:")
    print(outputs[0]['generated_text'].replace(alpaca_prompt, "").strip())
    print("="*60)

    # 메모리 정리
    del model, tokenizer, generator
    clear_memory()


# =====================================================================
# 2. Vicuna 모델 공략: 파이썬 알고리즘 코드 생성 (Code Generation)
# =====================================================================
def run_vicuna_task():
    # LMSYS 공식 Vicuna v1.5 모델 (상업적 이용이 가능한 LLaMA-2 기반 파생형)
    model_id = "lmsys/vicuna-7b-v1.5"
    print(f"\n[2/3] Vicuna 모델 로딩 중... (응용 분야: 코드 생성)")

    tokenizer = AutoTokenizer.from_pretrained(model_id)
    model = AutoModelForCausalLM.from_pretrained(
        model_id, torch_dtype=torch.float16, device_map="auto"
    )

    # Vicuna 고유의 USER/ASSISTANT 프롬프트 템플릿 구성 (ShareGPT 스타일)
    vicuna_prompt = (
        "A chat between a curious user and an artificial intelligence assistant. "
        "The assistant gives helpful, detailed, and polite answers to the user's questions.\n\n"
        "USER: Write a Python function that checks if a given string is a palindrome. "
        "Include a brief docstring and one example test case.\n"
        "ASSISTANT:\n"
    )

    generator = pipeline("text-generation", model=model, tokenizer=tokenizer)
    outputs = generator(vicuna_prompt, max_new_tokens=300, do_sample=False) # 코드는 일관성을 위해 샘플링 끔

    print("\n💻 [Vicuna 결과] 파이썬 코드 생성:")
    print(outputs[0]['generated_text'].replace(vicuna_prompt, "").strip())
    print("="*60)

    # 메모리 정리
    del model, tokenizer, generator
    clear_memory()


# =====================================================================
# 3. Koala 모델 공략: 고객 리뷰 감성 분석 (Sentiment Analysis/Classification)
# =====================================================================
def run_koala_task():
    # UC 버클리 Koala 구조를 재현한 오픈소스 체크포인트
    model_id = "TheBloke/koala-7B-HF"
    print(f"\n[3/3] Koala 모델 로딩 중... (응용 분야: 감성 및 텍스트 분류)")

    tokenizer = AutoTokenizer.from_pretrained(model_id)
    model = AutoModelForCausalLM.from_pretrained(
        model_id, torch_dtype=torch.float16, device_map="auto"
    )

    # Koala 고유의 BEGINNING OF CHAT 대화 구조 템플릿 구성
    koala_prompt = (
        "BEGINNING OF CHAT "
        "USER: Classify the sentiment of the following product review as either POSITIVE, NEGATIVE, or NEUTRAL. "
        "Provide a one-word answer and a very brief reason.\n"
        "Review: 'I bought this microphone last week. The audio clarity is amazing, but the XLR cable included in the box was defective, so I had to buy a replacement.'\n"
        "GPT:\n"
    )

    generator = pipeline("text-generation", model=model, tokenizer=tokenizer)
    outputs = generator(koala_prompt, max_new_tokens=150, do_sample=True, temperature=0.5)

    print("\n📊 [Koala 결과] 고객 리뷰 감성 분석:")
    print(outputs[0]['generated_text'].replace(koala_prompt, "").strip())
    print("="*60)

    # 메모리 정리
    del model, tokenizer, generator
    clear_memory()


# =====================================================================
# 메인 실행 제어
# =====================================================================
if __name__ == "__main__":
    if not torch.cuda.is_available():
        print("이 코드는 거대한 LLM들을 연속으로 구동하므로 고성능 GPU(VRAM 16GB 이상) 환경이 필요합니다.")

    # 순차적으로 모델을 로드하여 고유 태스크 수행 후 메모리 해제
    run_alpaca_task()
    run_vicuna_task()
    run_koala_task()
    print("\n모든 모델의 응용 태스크 처리가 완료되었습니다.")

```
